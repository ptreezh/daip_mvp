### 1. 健壮性模型设计
```python
class PermissionInteraction(BaseModel):
    """健壮权限交互模型"""
    request_id: str = Field(default_factory=lambda: f"perm_{uuid.uuid4().hex[:16]}")
    tool_name: str
    args: Dict[str, Any]
    state: PermissionState = Field(default=PermissionState.PENDING)
    response: Optional[PermissionResponse] = None
    timeout_seconds: float = Field(default=30.0)  # 超时保护
    error_count: int = Field(default=0)           # 错误计数
    retry_count: int = Field(default=0)           # 重试计数
    max_retries: int = Field(default=3)           # 最大重试
    circuit_breaker_open: bool = Field(default=False)  # 断路器
    is_duplicate: bool = Field(default=False)     # 重复检测
    
    # 健壮性方法
    def is_expired(self) -> bool:                 # 过期检测
    def can_retry(self) -> bool:                  # 重试逻辑
    def mark_as_duplicate(self) -> None:          # 重复处理
    def increment_error_count(self) -> None:     # 错误统计
```
