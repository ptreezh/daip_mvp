## 使用示例

```python
# 创建降级管理器
registry = AlgorithmRegistry()
selector = AlgorithmSelector(registry)
config = FallbackConfig(
    max_fallback_depth=3,
    failure_threshold=5,
    retry_strategy=RetryStrategy.EXPONENTIAL_BACKOFF
)
fallback_manager = FallbackManager(registry, selector, config)
utils = FallbackManagerUtils(fallback_manager)

# 获取降级链
failure_context = FailureContext(
    failed_algorithm="failed_algo",
    error_type="RuntimeError",
    error_message="Algorithm failed",
    execution_time=2.0
)
fallback_chain = fallback_manager.get_fallback_chain(
    "failed_algo", request, failure_context
)

# 执行降级
response = await fallback_manager.execute_fallback(
    fallback_chain[0], request, failure_context, fallback_depth=1
)

# 获取统计信息
stats = utils.get_fallback_stats()
reliability = utils.get_algorithm_reliability("algorithm_id")
```
