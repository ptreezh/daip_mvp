上下文清晰性检测目标

确保每个任务都有清晰的背景、目标和执行路径，避免开发过程中的理解偏差。

上下文完整性

  • 任务背景清晰
  • 相关依赖明确
  • 影响范围确定

目标明确性

  • 任务目标具体可衡量
  • 验收标准清晰
  • 成功指标量化

执行路径清晰

  • 实施步骤明确
  • 资源需求清楚
  • 风险识别充分

任务描述模板标准化

建立标准化的任务描述模板，确保任务信息的完整性和一致性。

完整任务模板

任务描述模板 
## 任务ID: [TASK-XXX]
### 任务标题: [简明扼要的任务标题]
### 任务类型: [功能开发|缺陷修复|技术优化|文档编写|测试实施]
### 优先级: [紧急|高|中|低]
### 状态: [待办|进行中|代码审查|测试中|已完成]

### 背景信息:
[描述为什么要执行这个任务，解决什么问题，相关的业务场景]

### 相关需求:
- [关联的需求ID和简要说明]
- [上游任务ID]
- [并行任务ID]

### 任务目标:
[明确描述任务要达成的具体目标，使用SMART原则]

### 验收标准:
- [具体、可验证的验收条件1]
- [具体、可验证的验收条件2]
- [具体、可验证的验收条件3]

### 实施步骤:
1. [第一步详细说明]
2. [第二步详细说明]
3. [第三步详细说明]

### 资源需求:
- [人力需求：开发人员、测试人员等]
- [技术资源：服务器、数据库、第三方服务等]
- [时间预算：预计工时]

### 风险评估:
- [技术风险及应对措施]
- [时间风险及缓冲方案]
- [质量风险及保障措施]

### 交付物:
- [代码实现]
- [测试用例]
- [文档更新]
- [部署脚本]

// 上下文完整性检查
function checkContextCompleteness(task) {
  const errors = [];
  const warnings = [];
  
  // 检查背景信息
  if (!task.background || task.background.trim().length < 20) {
    errors.push('背景信息过于简短或缺失');
  }
  
  // 检查相关需求
  if (!task.relatedRequirements || task.relatedRequirements.length === 0) {
    warnings.push('未关联相关需求，可能导致开发偏离目标');
  }
  
  // 检查依赖关系
  if (task.dependencies && task.dependencies.length > 0) {
    task.dependencies.forEach(dep => {
      if (!dep.id || !dep.description) {
        warnings.push('依赖关系描述不完整');
      }
    });
  }
  
  return {
    valid: errors.length === 0,
    errors,
    warnings
  };
}

// 目标明确性检查
function checkGoalClarity(task) {
  const errors = [];
  
  // 检查任务目标
  if (!task.goal || task.goal.trim().length < 15) {
    errors.push('任务目标描述不清晰');
  } else {
    // 检查是否符合SMART原则
    const smartChecks = {
      Specific: /[具体|明确|详细]/i.test(task.goal),
      Measurable: /[数量|指标|百分比|时间]/i.test(task.goal),
      Achievable: !/[不可能|很难|或许]/i.test(task.goal),
      Relevant: /[业务|用户|系统]/i.test(task.goal),
      Timebound: /[天|周|月|完成|截止]/i.test(task.goal)
    };
    
    const missingSmart = Object.keys(smartChecks).filter(key => 
      !smartChecks[key]
    );
    if (missingSmart.length > 2) {
      errors.push(`任务目标缺少SMART要素: ${missingSmart.join(', ')}`);
    }
  }
  
  // 检查验收标准
  if (!task.acceptanceCriteria || task.acceptanceCriteria.length === 0) {
    errors.push('缺少验收标准');
  } else {
    task.acceptanceCriteria.forEach((criteria, index) => {
      if (!criteria || criteria.trim().length === 0) {
        errors.push(`验收标准第${index + 1}项为空`);
      } else if (!/[应该|必须|能够|可以]/.test(criteria)) {
        errors.push(`验收标准第${index + 1}项缺少行为动词`);
      }
    });
  }
  
  return {
    valid: errors.length === 0,
    errors
  };
}

// 执行路径清晰度检查
function checkExecutionPathClarity(task) {
  const errors = [];
  const warnings = [];
  
  // 检查实施步骤
  if (!task.steps || task.steps.length === 0) {
    errors.push('缺少实施步骤');
  } else {
    if (task.steps.length < 2) {
      warnings.push('实施步骤过少，可能不够详细');
    }
    
    task.steps.forEach((step, index) => {
      if (!step || step.trim().length === 0) {
        errors.push(`实施步骤第${index + 1}项为空`);
      } else if (step.length < 10) {
        warnings.push(`实施步骤第${index + 1}项描述过于简单`);
      }
    });
  }
  
  return {
    valid: errors.length === 0,
    errors,
    warnings
  };
}

class TaskClarityChecker {
  constructor() {
    this.checkRules = [
      this.checkStructure,
      this.checkContext,
      this.checkGoal,
      this.checkExecutionPath
    ];
  }
  
  // 检查任务结构完整性
  checkStructure(task) {
    const requiredFields = [
      'id', 'title', 'type', 'priority', 'background', 
      'goal', 'acceptanceCriteria', 'steps'
    ];
    
    const missingFields = requiredFields.filter(field => 
      !task[field] || 
      (Array.isArray(task[field]) && task[field].length === 0) ||
      (typeof task[field] === 'string' && task[field].trim().length === 0)
    );
    
    if (missingFields.length > 0) {
      return {
        rule: '结构完整性检查',
        passed: false,
        message: `缺少必要字段: ${missingFields.join(', ')}`
      };
    }
    
    return {
      rule: '结构完整性检查',
      passed: true,
      message: '任务结构完整'
    };
  }
  
  // 检查上下文完整性
  checkContext(task) {
    const errors = [];
    const warnings = [];
    
    // 检查背景信息
    if (!task.background || task.background.trim().length < 20) {
      errors.push('背景信息过于简短或缺失');
    }
    
    // 检查相关需求
    if (!task.relatedRequirements || task.relatedRequirements.length === 0) {
      warnings.push('未关联相关需求');
    }
    
    // 检查依赖关系
    if (task.dependencies && task.dependencies.length > 5) {
      warnings.push(`依赖过多 (${task.dependencies.length}个)，建议拆分任务`);
    }
    
    if (errors.length > 0) {
      return {
        rule: '上下文完整性检查',
        passed: false,
        message: errors.join('; '),
        warnings: warnings
      };
    }
    
    const message = warnings.length > 0 
      ? `存在警告: ${warnings.join('; ')}` 
      : '上下文信息完整';
    
    return {
      rule: '上下文完整性检查',
      passed: true,
      message: message,
      warnings: warnings
    };
  }
  
  // 检查目标明确性
  checkGoal(task) {
    const errors = [];
    
    // 检查任务目标
    if (!task.goal || task.goal.trim().length < 15) {
      errors.push('任务目标描述不清晰');
    } else {
      // 检查是否符合SMART原则
      const hasSpecific = /[具体|明确|详细]/i.test(task.goal);
      const hasMeasurable = /[数量|指标|百分比|时间]/i.test(task.goal);
      const hasAchievable = !/[不可能|很难|或许]/i.test(task.goal);
      const hasRelevant = /[业务|用户|系统]/i.test(task.goal);
      const hasTimebound = /[天|周|月|完成|截止]/i.test(task.goal);
      
      const missingElements = [];
      if (!hasSpecific) missingElements.push('具体性');
      if (!hasMeasurable) missingElements.push('可衡量性');
      if (!hasAchievable) missingElements.push('可达成性');
      if (!hasRelevant) missingElements.push('相关性');
      if (!hasTimebound) missingElements.push('时限性');
      
      if (missingElements.length > 2) {
        errors.push(`任务目标缺少SMART要素: ${missingElements.join(', ')}`);
      }
    }
    
    // 检查验收标准
    if (!task.acceptanceCriteria || task.acceptanceCriteria.length === 0) {
      errors.push('缺少验收标准');
    } else {
      task.acceptanceCriteria.forEach((criteria, index) => {
        if (!criteria || criteria.trim().length === 0) {
          errors.push(`验收标准第${index + 1}项为空`);
        } else if (!/[应该|必须|能够|可以]/.test(criteria)) {
          errors.push(`验收标准第${index + 1}项缺少行为动词`);
        }
      });
    }
    
    if (errors.length > 0) {
      return {
        rule: '目标明确性检查',
        passed: false,
        message: errors.join('; ')
      };
    }
    
    return {
      rule: '目标明确性检查',
      passed: true,
      message: '任务目标明确'
    };
  }
  
  // 检查执行路径清晰度
  checkExecutionPath(task) {
    const errors = [];
    const warnings = [];
    
    // 检查实施步骤
    if (!task.steps || task.steps.length === 0) {
      errors.push('缺少实施步骤');
    } else {
      if (task.steps.length < 2) {
        warnings.push('实施步骤过少，可能不够详细');
      }
      
      task.steps.forEach((step, index) => {
        if (!step || step.trim().length === 0) {
          errors.push(`实施步骤第${index + 1}项为空`);
        } else if (step.length < 10) {
          warnings.push(`实施步骤第${index + 1}项描述过于简单`);
        }
        
        // 检查步骤是否过于宽泛
        const broadKeywords = ['实现', '开发', '完成', '处理'];
        const isBroad = broadKeywords.some(keyword => 
          new RegExp(`^${keyword}|${keyword}$`).test(step.trim())
        );
        
        if (isBroad) {
          warnings.push(`步骤${index + 1}过于宽泛: "${step}"`);
        }
      });
    }
    
    if (errors.length > 0) {
      return {
        rule: '执行路径清晰度检查',
        passed: false,
        message: errors.join('; '),
        warnings: warnings
      };
    }
    
    const message = warnings.length > 0 
      ? `存在警告: ${warnings.join('; ')}` 
      : '执行路径清晰';
    
    return {
      rule: '执行路径清晰度检查',
      passed: true,
      message: message,
      warnings: warnings
    };
  }
  
  // 执行单项检查
  checkTask(task) {
    const results = [];
    
    for (const rule of this.checkRules) {
      try {
        const result = rule.call(this, task);
        results.push(result);
      } catch (error) {
        results.push({
          rule: rule.name,
          passed: false,
          message: `检查出错: ${error.message}`
        });
      }
    }
    
    return results;
  }
  
  // 执行全套检查
  checkAllTasks(tasks) {
    const report = {
      total: tasks.length,
      passed: 0,
      failed: 0,
      warnings: 0,
      details: []
    };
    
    for (const task of tasks) {
      const checkResults = this.checkTask(task);
      const taskPassed = checkResults.every(result => result.passed);
      const hasWarnings = checkResults.some(result => result.warnings && result.warnings.length > 0);
      
      if (taskPassed) {
        report.passed++;
      } else {
        report.failed++;
      }
      
      if (hasWarnings) {
        report.warnings++;
      }
      
      report.details.push({
        taskId: task.id,
        taskTitle: task.title,
        passed: taskPassed,
        hasWarnings: hasWarnings,
        checks: checkResults
      });
    }
    
    return report;
  }
  
  // 生成检测报告
  generateReport(report) {
    let reportText = `任务分解上下文清晰性检测报告\n`;
    reportText += `================================\n\n`;
    reportText += `检测时间: ${new Date().toLocaleString()}\n`;
    reportText += `总任务数量: ${report.total}\n`;
    reportText += `通过检测: ${report.passed}\n`;
    reportText += `存在问题: ${report.failed}\n`;
    reportText += `存在警告: ${report.warnings}\n\n`;
    
    if (report.failed > 0) {
      reportText += `❌ 存在问题的任务数量: ${report.failed}\n\n`;
      reportText += `详细问题列表:\n`;
      
      for (const detail of report.details) {
        if (!detail.passed) {
          reportText += `\n任务ID: ${detail.taskId}\n`;
          reportText += `任务标题: ${detail.taskTitle}\n`;
          
          for (const check of detail.checks) {
            if (!check.passed) {
              reportText += `  - ${check.rule}: ${check.message}\n`;
            }
          }
        }
      }
    }
    
    if (report.warnings > 0) {
      reportText += `\n⚠️ 存在警告的任务数量: ${report.warnings}\n\n`;
      reportText += `警告详情:\n`;
      
      for (const detail of report.details) {
        if (detail.hasWarnings) {
          reportText += `\n任务ID: ${detail.taskId}\n`;
          reportText += `任务标题: ${detail.taskTitle}\n`;
          
          for (const check of detail.checks) {
            if (check.warnings && check.warnings.length > 0) {
              reportText += `  - ${check.rule}警告: ${check.warnings.join('; ')}\n`;
            }
          }
        }
      }
    }
    
    if (report.failed === 0 && report.warnings === 0) {
      reportText += `✅ 所有任务均通过清晰性检测\n`;
    }
    
    return reportText;
  }
}
