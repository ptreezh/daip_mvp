模块概述

模块职责

P5智能体引擎是DAIP-LIVE系统的核心执行引擎，负责意图识别、任务规划、步骤执行和结果整合。它采用事件驱动的异步架构，支持多层次的任务分解和实时用户交互控制。

核心功能

  • 意图识别：理解用户输入，识别执行意图和所需技能
  • 任务规划：将复杂任务分解为可执行的步骤序列
  • 步骤执行：逐步执行任务，支持实时监控和干预
  • 上下文管理：维护对话上下文和执行状态
  • 事件驱动：基于事件的异步执行机制
  • 技能调度：动态调用和管理各种技能工具

架构设计

核心组件

1. EnhancedChatExecutor

主要的聊天执行器，负责协调整个对话和任务执行流程，管理执行状态和用户交互。

2. IntentRecognizer

意图识别器，分析用户输入，识别用户的意图、需求和期望的操作类型。

3. StepExecutor

步骤执行器，负责执行具体任务步骤，支持异步执行和实时状态更新。

4. ContextManager

上下文管理器，维护对话历史、任务状态和执行上下文信息。

5. EventSystem

事件系统，实现组件间的松耦合通信，支持事件的发布和订阅机制。

class EventDrivenExecutor: """基于事件驱动的执行器""" async def execute_with_events(self, task: Task) -> ExecutionResult: # 发布任务开始事件 await self.event_bus.publish(EventType.TASK_STARTED, task) try: # 执行任务步骤 for step in task.steps: await self.event_bus.publish(EventType.STEP_STARTED, step) result = await self.execute_step(step) await self.event_bus.publish(EventType.STEP_COMPLETED, { 'step': step, 'result': result }) # 发布任务完成事件 await self.event_bus.publish(EventType.TASK_COMPLETED, task) except Exception as e: await self.event_bus.publish(EventType.TASK_FAILED, { 'task': task, 'error': str(e) }) raise

SOLID原则的具体体现

S - 单一职责原则 (Single Responsibility Principle)

class IntentRecognizer: """只负责意图识别""" async def recognize_intent(self, user_input: str) -> Intent: """专门处理意图识别逻辑""" pass class StepExecutor: """只负责步骤执行""" async def execute_step(self, step: Step) -> StepResult: """专门处理步骤执行逻辑""" pass class ContextManager: """只负责上下文管理""" def update_context(self, new_data: Dict) -> None: """专门处理上下文更新""" pass class EventSystem: """只负责事件管理""" async def publish(self, event_type: str, data: Any) -> None: """专门处理事件发布""" pass

O - 开放封闭原则 (Open-Closed Principle)

from abc import ABC, abstractmethod class StepHandler(ABC): """步骤处理器基类""" @abstractmethod async def handle(self, step: Step) -> StepResult: pass # 新增步骤处理器无需修改现有代码 class FileOperationHandler(StepHandler): """文件操作处理器""" async def handle(self, step: Step) -> StepResult: # 实现文件操作逻辑 pass class WebSearchHandler(StepHandler): """网络搜索处理器""" async def handle(self, step: Step) -> StepResult: # 实现网络搜索逻辑 pass class StepExecutor: def __init__(self): self.handlers = {} # 处理器注册表 def register_handler(self, step_type: str, handler: StepHandler): """注册新的处理器，无需修改现有代码""" self.handlers[step_type] = handler async def execute_step(self, step: Step) -> StepResult: handler = self.handlers.get(step.type) if handler: return await handler.handle(step) raise ValueError(f"No handler for step type: {step.type}")

L - 里氏替换原则 (Liskov Substitution Principle)

class BaseContextManager(ABC): """上下文管理器基类""" @abstractmethod async def get_context(self) -> Dict[str, Any]: pass @abstractmethod async def update_context(self, updates: Dict[str, Any]) -> None: pass class MemoryContextManager(BaseContextManager): """内存上下文管理器""" async def get_context(self) -> Dict[str, Any]: return self._memory_context async def update_context(self, updates: Dict[str, Any]) -> None: self._memory_context.update(updates) class DatabaseContextManager(BaseContextManager): """数据库上下文管理器""" async def get_context(self) -> Dict[str, Any]: return await self._db.load_context() async def update_context(self, updates: Dict[str, Any]) -> None: await self._db.save_context(updates) # 任何BaseContextManager的子类都可以替换使用 executor = ChatExecutor(MemoryContextManager()) # ✓ 正确 executor = ChatExecutor(DatabaseContextManager()) # ✓ 正确

I - 接口隔离原则 (Interface Segregation Principle)

接口拆分示例 Python 
# 将大接口拆分为多个小接口
class IntentHandler(ABC):
    @abstractmethod
    async def handle_intent(self, intent: Intent) -> None:
        pass

class TaskPlanner(ABC):
    @abstractmethod
    async def plan_task(self, intent: Intent) -> Task:
        pass

class StepExecutor(ABC):
    @abstractmethod
    async def execute_step(self, step: Step) -> StepResult:
        pass

class ProgressReporter(ABC):
    @abstractmethod
    async def report_progress(self, progress: float) -> None:
        pass

# 客户端只依赖需要的接口
class SimpleExecutor:
    """简单执行器，只需要执行功能"""
    def __init__(self, step_executor: StepExecutor):
        self.step_executor = step_executor

class InteractiveExecutor:
    """交互式执行器，需要进度报告"""
    def __init__(self,
                 step_executor: StepExecutor,
                 progress_reporter: ProgressReporter):
        self.step_executor = step_executor
        self.progress_reporter = progress_reporter

import asyncio from typing import AsyncIterator, Dict, Any, List from dataclasses import dataclass from enum import Enum class ExecutionState(Enum): IDLE = "idle" PLANNING = "planning" EXECUTING = "executing" COMPLETED = "completed" FAILED = "failed" @dataclass class ExecutionContext: user_input: str intent: 'Intent' = None task: 'Task' = None steps_completed: List['StepResult'] = None current_step: int = 0 state: ExecutionState = ExecutionState.IDLE metadata: Dict[str, Any] = None class EnhancedChatExecutor: """增强型聊天执行器""" def __init__(self, intent_recognizer: IntentRecognizer, task_planner: TaskPlanner, step_executor: StepExecutor, context_manager: ContextManager): self.intent_recognizer = intent_recognizer self.task_planner = task_planner self.step_executor = step_executor self.context_manager = context_manager self.execution_context: Optional[ExecutionContext] = None async def process_input(self, user_input: str) -> AsyncIterator[str]: """处理用户输入的主要入口点""" try: # 创建执行上下文 self.execution_context = ExecutionContext(user_input=user_input) # 阶段1：意图识别 yield "正在理解您的需求..." await self._recognize_intent() # 阶段2：任务规划 yield "正在制定执行计划..." await self._plan_task() # 阶段3：执行任务 async for progress in self._execute_task(): yield progress except Exception as e: yield f"执行过程中发生错误：{str(e)}" self.execution_context.state = ExecutionState.FAILED async def _recognize_intent(self) -> None: """识别用户意图""" self.execution_context.state = ExecutionState.PLANNING # 结合上下文进行意图识别 context = await self.context_manager.get_context() self.execution_context.intent = await self.intent_recognizer.recognize_intent( self.execution_context.user_input, context ) async def _plan_task(self) -> None: """制定任务计划""" self.execution_context.task = await self.task_planner.plan_task( self.execution_context.intent, await self.context_manager.get_context() ) async def _execute_task(self) -> AsyncIterator[str]: """执行任务步骤""" self.execution_context.state = ExecutionState.EXECUTING for i, step in enumerate(self.execution_context.task.steps): self.execution_context.current_step = i # 执行步骤 yield f"正在执行步骤 {i+1}/{len(self.execution_context.task.steps)}: {step.description}" try: result = await self.step_executor.execute_step(step) self.execution_context.steps_completed.append(result) # 更新上下文 await self.context_manager.update_context({ 'last_step_result': result, 'execution_progress': (i + 1) / len(self.execution_context.task.steps) }) yield f"✓ {step.description} - {result.message}" except Exception as e: yield f"✗ {step.description} - 错误: {str(e)}" raise self.execution_context.state = ExecutionState.COMPLETED yield "任务执行完成！" async def get_execution_status(self) -> Dict[str, Any]: """获取执行状态""" if not self.execution_context: return {'state': 'idle'} return { 'state': self.execution_context.state.value, 'progress': (self.execution_context.current_step / len(self.execution_context.task.steps) if self.execution_context.task else 0), 'current_step': self.execution_context.current_step, 'total_steps': len(self.execution_context.task.steps) if self.execution_context.task else 0 }

from typing import Dict, Callable import asyncio class EventType: STEP_STARTED = "step_started" STEP_COMPLETED = "step_completed" STEP_FAILED = "step_failed" PROGRESS_UPDATED = "progress_updated" class EventSystem: """事件系统""" def __init__(self): self._subscribers: Dict[str, List[Callable]] = {} def subscribe(self, event_type: str, callback: Callable): """订阅事件""" if event_type not in self._subscribers: self._subscribers[event_type] = [] self._subscribers[event_type].append(callback) async def publish(self, event_type: str, data: Any): """发布事件""" if event_type in self._subscribers: tasks = [callback(data) for callback in self._subscribers[event_type]] await asyncio.gather(*tasks, return_exceptions=True) class EventDrivenStepExecutor: """基于事件的步骤执行器""" def __init__(self, event_system: EventSystem): self.event_system = event_system self._running_tasks: Dict[str, asyncio.Task] = {} async def execute_step(self, step: Step) -> StepResult: """执行步骤并发布事件""" step_id = f"step_{step.id}_{time.time()}" try: # 发布步骤开始事件 await self.event_system.publish(EventType.STEP_STARTED, { 'step_id': step_id, 'step': step, 'timestamp': time.time() }) # 执行步骤 result = await self._execute_step_logic(step) # 发布步骤完成事件 await self.event_system.publish(EventType.STEP_COMPLETED, { 'step_id': step_id, 'step': step, 'result': result, 'timestamp': time.time() }) return result except Exception as e: # 发布步骤失败事件 await self.event_system.publish(EventType.STEP_FAILED, { 'step_id': step_id, 'step': step, 'error': str(e), 'timestamp': time.time() }) raise async def _execute_step_logic(self, step: Step) -> StepResult: """具体的步骤执行逻辑""" if step.type == "file_operation": return await self._handle_file_operation(step) elif step.type == "web_search": return await self._handle_web_search(step) elif step.type == "api_call": return await self._handle_api_call(step) else: raise ValueError(f"Unknown step type: {step.type}") async def _handle_file_operation(self, step: Step) -> StepResult: """处理文件操作""" operation = step.parameters.get('operation') file_path = step.parameters.get('path') if operation == 'create': # 创建文件逻辑 with open(file_path, 'w') as f: f.write(step.parameters.get('content', '')) return StepResult( success=True, message=f"文件 {file_path} 创建成功", data={'file_path': file_path} ) elif operation == 'read': # 读取文件逻辑 with open(file_path, 'r') as f: content = f.read() return StepResult( success=True, message=f"文件 {file_path} 读取成功", data={'content': content} ) # 其他文件操作... async def _handle_web_search(self, step: Step) -> StepResult: """处理网络搜索""" query = step.parameters.get('query') # 实现网络搜索逻辑 # 这里可以调用搜索引擎API return StepResult( success=True, message=f"搜索完成: {query}", data={'search_results': []} ) async def _handle_api_call(self, step: Step) -> StepResult: """处理API调用""" api_url = step.parameters.get('url') method = step.parameters.get('method', 'GET') # 实现API调用逻辑 return StepResult( success=True, message=f"API调用完成: {method} {api_url}", data={'response': {}} )

3. 上下文感知的意图识别器

from typing import Optional, List import re @dataclass class Intent: type: str action: Optional[str] = None entities: Dict[str, Any] = None confidence: float = 0.0 metadata: Dict[str, Any] = None class ContextAwareIntentRecognizer: """上下文感知的意图识别器""" def __init__(self): self.intent_patterns = self._load_intent_patterns() self.context_weights = { 'recent_history': 0.3, 'session_context': 0.2, 'user_preferences': 0.1 } async def recognize_intent(self, user_input: str, context: Dict[str, Any] = None) -> Intent: """识别用户意图，考虑上下文信息""" context = context or {} # 基础模式匹配 base_intent = self._pattern_matching(user_input) # 上下文增强 enhanced_intent = await self._enhance_with_context(base_intent, context) # 置信度计算 enhanced_intent.confidence = self._calculate_confidence(enhanced_intent, user_input, context) return enhanced_intent def _pattern_matching(self, user_input: str) -> Intent: """基于模式的意图识别""" normalized_input = user_input.lower().strip() for intent_type, patterns in self.intent_patterns.items(): for pattern, action in patterns: match = re.search(pattern, normalized_input, re.IGNORECASE) if match: return self._create_intent_from_match(intent_type, action, match, user_input) return Intent(type="unknown", confidence=0.0) async def _enhance_with_context(self, base_intent: Intent, context: Dict[str, Any]) -> Intent: """使用上下文信息增强意图识别""" # 检查最近的对话历史 recent_history = context.get('recent_history', []) if recent_history: last_intent = recent_history[-1].get('intent') if self._is_follow_up_intent(base_intent, last_intent): base_intent.type = last_intent.type base_intent.action = 'follow_up' # 检查会话上下文 session_context = context.get('session_context', {}) if 'current_project' in session_context: base_intent.entities = base_intent.entities or {} base_intent.entities['project_context'] = session_context['current_project'] # 检查用户偏好 user_preferences = context.get('user_preferences', {}) if 'default_action' in user_preferences and base_intent.type == "unknown": base_intent.type = "default_action" base_intent.action = user_preferences['default_action'] return base_intent def _is_follow_up_intent(self, current_intent: Intent, previous_intent: Optional[Intent]) -> bool: """判断是否为后续意图""" follow_up_indicators = ['继续', '然后', '接下来', '还有', '再'] current_text = current_intent.metadata.get('original_text', '').lower() if current_intent.metadata else '' return (previous_intent and any(indicator in current_text for indicator in follow_up_indicators)) def _calculate_confidence(self, intent: Intent, user_input: str, context: Dict[str, Any]) -> float: """计算意图识别的置信度""" confidence = 0.5 # 基础置信度 # 模式匹配置信度 if intent.type != "unknown": confidence += 0.3 # 上下文一致性 recent_history = context.get('recent_history', []) if recent_history and intent.type == recent_history[-1].get('intent', {}).get('type'): confidence += 0.2 # 用户输入的明确性 if len(user_input.split()) > 3: # 输入较详细 confidence += 0.1 return min(confidence, 1.0) def _create_intent_from_match(self, intent_type: str, action: str, match, original_text: str) -> Intent: """从正则匹配创建意图对象""" entities = {} if intent_type == "file_operation": entities['file'] = True if match.groups(): entities['filename'] = match.groups()[0] elif intent_type == "web_search": if match.groups(): entities['search_query'] = match.groups()[0] return Intent( type=intent_type, action=action, entities=entities, metadata={'original_text': original_text} ) def _load_intent_patterns(self) -> Dict[str, List]: """加载意图模式""" return { "file_operation": [ (r"创建.*文件[：:]\s*(.+)", "create"), (r"删除.*文件[：:]\s*(.+)", "delete"), (r"编辑.*文件[：:]\s*(.+)", "edit"), (r"读取.*文件[：:]\s*(.+)", "read") ], "web_search": [ (r"搜索\s*(.+)", "search"), (r"查找\s*(.+)", "search"), (r"搜索关于(.+)的信息", "search") ], "code_execution": [ (r"运行\s*(.+)", "run"), (r"执行\s*(.+)", "execute"), (r"编译\s*(.+)", "compile") ], "system_command": [ (r"安装\s*(.+)", "install"), (r"配置\s*(.+)", "configure"), (r"设置\s*(.+)", "setup") ] }

最佳实践与规范

KISS原则应用

  • 简洁的接口设计：核心接口只包含必要的方法，避免过度复杂
  • 清晰的执行流程：线性化的执行步骤，易于理解和调试
  • 最小化状态管理：只在必要的地方维护状态，避免状态复杂性

DRY原则应用

  • 公共执行器基类：提取所有执行器的公共逻辑
  • 可复用的步骤处理器：步骤处理器可以在不同执行器中复用
  • 统一的错误处理：统一的异常处理和错误恢复机制

YAGNI原则应用

  • 按需功能实现：只实现当前需要的功能特性
  • 渐进式复杂度：从简单的执行器开始，逐步增加复杂功能
  • 避免过度抽象：不需要的抽象层会带来不必要的复杂性

class EventEmitter:
    """事件发射器基类"""

    def __init__(self):
        self._event_handlers = defaultdict(list)
        self._loop = asyncio.get_event_loop()

    def on(self, event: str, handler: Callable):
        """注册事件处理器"""
        self._event_handlers[event].append(handler)

    def emit(self, event: str, data: Any = None):
        """异步发射事件"""
        if event in self._event_handlers:
            for handler in self._event_handlers[event]:
                asyncio.create_task(handler(data))

    async def emit_sync(self, event: str, data: Any = None):
        """同步发射事件，等待所有处理器完成"""
        if event in self._event_handlers:
            tasks = [handler(data) for handler in self._event_handlers[event]]
            await asyncio.gather(*tasks, return_exceptions=True)

class ExecutionError(Exception):
    """执行错误基类"""
    def __init__(self, message: str, step: Step = None, cause: Exception = None):
        self.message = message
        self.step = step
        self.cause = cause
        super().__init__(message)

class ErrorRecoveryStrategy:
    """错误恢复策略"""

    @staticmethod
    async def retry_step(step: Step, max_retries: int = 3) -> StepResult:
        """重试策略"""
        for attempt in range(max_retries + 1):
            try:
                return await execute_step(step)
            except Exception as e:
                if attempt == max_retries:
                    raise ExecutionError(f"Step failed after {max_retries} retries", step, e)
                await asyncio.sleep(1)

    @staticmethod
    async def skip_step(step: Step) -> StepResult:
        """跳过策略"""
        return StepResult(
            success=True,
            message=f"Step skipped: {step.description}",
            skipped=True
        )

    @staticmethod
async def alternative_step(original_step: Step, alternative: Step) -> StepResult: """替代步骤策略""" try: return await execute_step(alternative) except Exception as e: raise ExecutionError(f"Alternative step also failed", alternative, e)
