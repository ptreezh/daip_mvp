模块概述

模块职责

P6终端界面模块为DAIP-LIVE系统提供用户交互界面，包括CLI（命令行界面）和TUI（文本用户界面）两种交互模式。它负责用户输入解析、命令路由、状态展示和交互反馈，为用户提供高效、直观的操作体验。

核心功能

  • CLI命令系统：基于Typer的命令行工具，支持脚本化操作
  • TUI交互界面：基于Textual的现代化终端UI，提供丰富的交互体验
  • 命令路由：智能命令解析和路由分发
  • 状态管理：界面状态和会话状态的统一管理
  • 实时反馈：异步操作进度显示和实时状态更新
  • 模块化设计：界面组件的模块化设计和复用

UI组件工厂，负责创建和管理TUI组件，支持组件复用和动态生成。

异步状态更新

from textual import events from textual.reactive import reactive import asyncio class AsyncStatusWidget(Static): """异步状态更新组件""" status_text: reactive[str] = reactive("Initializing...") progress: reactive[float] = reactive(0.0) def __init__(self, **kwargs): super().__init__(**kwargs) self._update_task = None def on_mount(self) -> None: """组件挂载时启动更新任务""" self._update_task = asyncio.create_task(self._update_status_loop()) def on_unmount(self) -> None: """组件卸载时清理任务""" if self._update_task: self._update_task.cancel() async def _update_status_loop(self): """状态更新循环""" while True: try: # 获取系统状态 status_info = await self._get_system_status() # 更新响应式属性 self.status_text = status_info["message"] self.progress = status_info["progress"] # 刷新显示 self.refresh() # 等待下次更新 await asyncio.sleep(1) except asyncio.CancelledError: break except Exception as e: self.status_text = f"Error: {str(e)}" self.refresh() await asyncio.sleep(5) async def _get_system_status(self) -> Dict[str, Any]: """获取系统状态信息""" # 这里可以调用实际的系统状态接口 return { "message": "System running normally", "progress": 0.75 } def render(self) -> str: """渲染组件内容""" return f""" [b]Status:[/b] {self.status_text} [b]Progress:[/b] {self.progress:.1%} {'█' * int(self.progress * 20)}{'░' * (20 - int(self.progress * 20))} """

SOLID原则的具体体现

S - 单一职责原则 (Single Responsibility Principle)

class CommandParser: """只负责命令解析""" def parse(self, input_text: str) -> ParsedCommand: """专门处理命令解析逻辑""" pass class CommandValidator: """只负责命令验证""" def validate(self, command: ParsedCommand) -> ValidationResult: """专门处理命令验证逻辑""" pass class CommandExecutor: """只负责命令执行""" async def execute(self, command: ParsedCommand) -> ExecutionResult: """专门处理命令执行逻辑""" pass class UIComponent: """只负责UI组件管理""" def render(self) -> str: """专门处理UI渲染逻辑""" pass def handle_event(self, event: UIEvent) -> None: """专门处理UI事件逻辑""" pass

O - 开放封闭原则 (Open-Closed Principle)

from abc import ABC, abstractmethod class CommandHandler(ABC): """命令处理器基类""" @abstractmethod async def handle(self, args: List[str]) -> CommandResult: pass @abstractmethod def get_help(self) -> str: pass # 新增命令处理器无需修改现有代码 class DebateCommandHandler(CommandHandler): """辩论命令处理器""" async def handle(self, args: List[str]) -> CommandResult: # 实现辩论命令逻辑 pass def get_help(self) -> str: return "Start and manage AI debates" class WikiCommandHandler(CommandHandler): """Wiki命令处理器""" async def handle(self, args: List[str]) -> CommandResult: # 实现Wiki命令逻辑 pass def get_help(self) -> str: return "Manage wiki pages and knowledge base" class CommandRouter: def __init__(self): self.handlers: Dict[str, CommandHandler] = {} def register_handler(self, command: str, handler: CommandHandler): """注册新的命令处理器，无需修改现有代码""" self.handlers[command] = handler async def route_command(self, command: str, args: List[str]) -> CommandResult: if command in self.handlers: return await self.handlers[command].handle(args) else: raise CommandNotFoundError(f"Unknown command: {command}")

class UIWidget(ABC):
    """UI组件基类"""

    @abstractmethod
    def render(self) -> str:
        pass

    @abstractmethod
    def handle_input(self, input_event: InputEvent) -> None:
        pass

    @abstractmethod
    def get_size(self) -> Size:
        pass

class ButtonWidget(UIWidget):
    """按钮组件"""

    def render(self) -> str:
        return "[Button]"

    def handle_input(self, input_event: InputEvent) -> None:
        if input_event.type == "click":
            self.on_click()

    def get_size(self) -> Size:
        return Size(width=10, height=1)

class TextInputWidget(UIWidget):
    """文本输入组件"""

    def render(self) -> str:
        return f"[Input: {self.text}]"

    def handle_input(self, input_event: InputEvent) -> None:
        if input_event.type == "key_press":
            self.update_text(input_event.key)

    def get_size(self) -> Size:
        return Size(width=20, height=1)

class TerminalInterface:
    """高层模块依赖抽象，不依赖具体实现"""

    def __init__(self,
                 command_parser: CommandParser,
                 ui_renderer: UIRenderer,
                 event_handler: EventHandler):
        self.command_parser = command_parser    # 依赖抽象
        self.ui_renderer = ui_renderer          # 依赖抽象
        self.event_handler = event_handler      # 依赖抽象

    async def run_interactive_session(self):
        """运行交互式会话"""
        while True:
            # 获取用户输入
            user_input = await self.get_user_input()

            # 解析命令
            command = await self.command_parser.parse(user_input)

            # 处理事件
            await self.event_handler.handle_command(command)

            # 更新界面
            await self.ui_renderer.update_display()

import re from typing import Dict, List, Optional @dataclass class ParsedCommand: action: str subcommand: Optional[str] = None arguments: List[str] = None options: Dict[str, Any] = None class CommandParser: """最小实现，让测试通过""" def __init__(self): self.command_patterns = { "debate": r"debate\s+(\w+)(?:\s+(.+))?", "search": r"search(?:\s+--(\w+)\s+(\w+))?(?:\s+(.+))?", "help": r"help\s*(\w+)?" } def parse(self, input_text: str) -> ParsedCommand: """最小实现，让测试通过""" input_text = input_text.strip() # 测试debate命令 debate_match = re.match(self.command_patterns["debate"], input_text) if debate_match: return ParsedCommand( action="start", subcommand="debate", arguments=[debate_match.group(2)] if debate_match.group(2) else [] ) # 测试search命令 search_match = re.match(self.command_patterns["search"], input_text) if search_match: options = {} if search_match.group(1) and search_match.group(2): options[search_match.group(1)] = search_match.group(2) return ParsedCommand( action="search", options=options, arguments=[search_match.group(3)] if search_match.group(3) else [] ) # 测试help命令 help_match = re.match(self.command_patterns["help"], input_text) if help_match: return ParsedCommand( action="help", arguments=[help_match.group(1)] if help_match.group(1) else [] ) raise CommandParseError(f"Unknown command: {input_text}")

class CommandParser: """重构后的命令解析器""" def __init__(self): self.command_handlers = { "debate": DebateCommandHandler(), "search": SearchCommandHandler(), "help": HelpCommandHandler(), "wiki": WikiCommandHandler(), "scaffold": ScaffoldCommandHandler() } self.option_parser = OptionParser() def parse(self, input_text: str) -> ParsedCommand: """完整的命令解析实现""" tokens = self._tokenize(input_text) if not tokens: raise CommandParseError("Empty command") # 识别主命令 main_command = tokens[0].lower() if main_command not in self.command_handlers: raise CommandParseError(f"Unknown command: {main_command}") # 获取对应的处理器 handler = self.command_handlers[main_command] # 解析选项和参数 options, arguments = self.option_parser.parse(tokens[1:]) # 使用处理器解析具体命令 return handler.parse_command(arguments, options) def _tokenize(self, input_text: str) -> List[str]: """命令分词""" # 处理引号包围的参数 import shlex return shlex.split(input_text) def register_command(self, command_name: str, handler: CommandHandler): """注册新的命令处理器""" self.command_handlers[command_name] = handler class OptionParser: """选项解析器""" def parse(self, tokens: List[str]) -> Tuple[Dict[str, Any], List[str]]: """解析命令行选项和参数""" options = {} arguments = [] i = 0 while i < len(tokens): token = tokens[i] if token.startswith("--"): # 长选项 if "=" in token: key, value = token[2:].split("=", 1) options[key] = value elif i + 1 < len(tokens) and not tokens[i + 1].startswith("--"): key = token[2:] value = tokens[i + 1] options[key] = value i += 1 else: options[token[2:]] = True elif token.startswith("-"): # 短选项 options[token[1:]] = True else: # 参数 arguments.append(token) i += 1 return options, arguments

前置验证 await self.validate(*args, **kwargs) # 权限检查 await self.check_permissions(*args, **kwargs) # 执行具体逻辑 result = await self.run(*args, **kwargs) return CLIResponse( success=True, message=f"Command {self.name} completed successfully", data=result ) except ValidationError as e: return CLIResponse( success=False, message=str(e), error_code="VALIDATION_ERROR" ) except PermissionError as e: return CLIResponse( success=False, message=str(e), error_code="PERMISSION_DENIED" ) except Exception as e: return CLIResponse( success=False, message=f"Error executing {self.name}: {str(e)}", error_code="EXECUTION_ERROR" ) async def validate(self, *args, **kwargs): """验证命令参数""" pass async def check_permissions(self, *args, **kwargs): """检查执行权限""" pass async def run(self, *args, **kwargs) -> Any: """执行具体命令逻辑""" raise NotImplementedError class DebateCommand(BaseCLICommand): """辩论命令实现""" def __init__(self, debate_manager): super().__init__("debate", "Manage AI debate sessions") self.debate_manager = debate_manager async def validate(self, *args, **kwargs): """验证辩论参数""" topic = kwargs.get('topic') if not topic or len(topic.strip()) < 10: raise ValidationError("Topic must be at least 10 characters long") async def run(self, *args, **kwargs) -> Dict[str, Any]: """执行辩论启动逻辑""" topic = kwargs['topic'] model = kwargs.get('model', 'gpt-3.5-turbo') rounds = kwargs.get('rounds', 3) roles = kwargs.get('roles', ['proponent', 'opponent']) # 启动辩论会话 session = await self.debate_manager.start_debate( topic=topic, model=model, rounds=rounds, roles=roles ) return { 'session_id': session.session_id, 'topic': session.topic, 'status': session.status.value } # CLI应用主类 class DAIPCLI: """DAIP-LIVE CLI应用""" def __init__(self): self.commands: Dict[str, BaseCLICommand] = {} self._register_default_commands() def _register_default_commands(self): """注册默认命令""" # 这里需要注入实际的依赖 self.register_command("debate", DebateCommand(debate_manager=None)) # 注册其他命令... def register_command(self, name: str, command: BaseCLICommand): """注册新命令""" self.commands[name] = command async def execute_command(self, command_name: str, **kwargs) -> CLIResponse: """执行命令""" if command_name not in self.commands: return CLIResponse( success=False, message=f"Unknown command: {command_name}", error_code="UNKNOWN_COMMAND" ) command = self.commands[command_name] return await command.execute(**kwargs) def get_command_help(self, command_name: str) -> Optional[str]: """获取命令帮助信息""" if command_name in self.commands: return self.commands[command_name].description return None def list_commands(self) -> List[str]: """列出所有可用命令""" return list(self.commands.keys()) # Typer集成 app = typer.Typer() # 全局CLI实例 daip_cli = DAIPCLI() @app.command() def debate( topic: str = typer.Argument(..., help="Debate topic"), model: Optional[str] = typer.Option(None, "--model", "-m", help="AI model"), rounds: int = typer.Option(3, "--rounds", "-r", help="Number of rounds") ): """Start an AI debate session""" import asyncio async def run_debate(): result = await daip_cli.execute_command( "debate", topic=topic, model=model, rounds=rounds ) if result.success: typer.echo(f"✓ {result.message}") if result.data: typer.echo(f"Session ID: {result.data['session_id']}") else: typer.echo(f"✗ {result.message}", err=True) raise typer.Exit(1) asyncio.run(run_debate()) if __name__ == "__main__": app()

2. 响应式TUI组件系统

from textual import events from textual.reactive import reactive from textual.widgets import Static, Button, Input from textual.containers import Container, Horizontal, Vertical from textual.app import ComposeResult import asyncio from typing import Any, Callable, Optional class ReactiveWidget(Static): """响应式UI组件基类""" data: reactive[Any] = reactive(None) loading: reactive[bool] = reactive(False) def __init__(self, **kwargs): super().__init__(**kwargs) self._update_callbacks = [] def watch_data(self, data: Any) -> None: """数据变化监听器""" self.refresh() def watch_loading(self, loading: bool) -> None: """加载状态变化监听器""" self.refresh() def add_update_callback(self, callback: Callable[[Any], None]): """添加数据更新回调""" self._update_callbacks.append(callback) async def update_data(self, new_data: Any): """更新数据并触发回调""" self.data = new_data # 执行所有回调 for callback in self._update_callbacks: if asyncio.iscoroutinefunction(callback): await callback(new_data) else: callback(new_data) class StatusBar(ReactiveWidget): """状态栏组件""" status_text: reactive[str] = reactive("Ready") progress: reactive[float] = reactive(0.0) def __init__(self, **kwargs): super().__init__(**kwargs) self._start_time = None def render(self) -> str: """渲染状态栏""" if self.loading: elapsed = self._get_elapsed_time() return f"[bold]Status:[/bold] {self.status_text} ({elapsed:.1f}s) [{self.progress:.0%}]" else: return f"[bold]Status:[/bold] {self.status_text}" def start_operation(self, message: str): """开始操作""" self.loading = True self.status_text = message self.progress = 0.0 self._start_time = asyncio.get_event_loop().time() def update_progress(self, progress: float, message: str = None): """更新进度""" self.progress = progress if message: self.status_text = message def finish_operation(self, message: str = "Operation completed"): """完成操作""" self.loading = False self.status_text = message self.progress = 1.0 self._start_time = None def _get_elapsed_time(self) -> float: """获取已用时间""" if self._start_time: return asyncio.get_event_loop().time() - self._start_time return 0.0 class CommandInput(ReactiveWidget): """命令输入组件""" current_input: reactive[str] = reactive("") command_history: reactive[list] = reactive([]) history_index: reactive[int] = reactive(-1) def compose(self) -> ComposeResult: """组合子组件""" yield Input(placeholder="Enter command...", id="command_input") def on_mount(self) -> None: """组件挂载时设置焦点""" self.query_one(Input).focus() def on_input_submitted(self, event: Input.Submitted) -> None: """处理输入提交""" command = event.value.strip() if command: # 添加到历史 self.command_history = self.command_history + [command] self.history_index = -1 # 触发命令执行事件 self.post_message(events.CommandEvent(command=command)) # 清空输入 self.current_input = "" event.input.value = "" class MainContentArea(ReactiveWidget): """主内容区域组件""" current_view: reactive[Optional[str]] = reactive(None) view_data: reactive[Dict[str, Any]] = reactive({}) def __init__(self, **kwargs): super().__init__(**kwargs) self._views = {} self._register_views() def _register_views(self): """注册视图""" self._views = { "chat": ChatView(), "debate": DebateView(), "wiki": WikiView(), "help": HelpView() } def switch_view(self, view_name: str, data: Dict[str, Any] = None): """切换视图""" if view_name not in self._views: raise ValueError(f"Unknown view: {view_name}") # 移除当前视图 if self.current_view: current_widget = self._views[self.current_view] if current_widget in self.children: self.remove_child(current_widget) # 设置新视图 self.current_view = view_name self.view_data = data or {} # 添加新视图 new_widget = self._views[view_name] self.mount(new_widget) def render(self) -> str: """渲染内容区域""" if not self.current_view: return "[dim]No view selected[/dim]" return f"Current view: {self.current_view}" class ModularTUIApp: """模块化TUI应用""" def __init__(self): self.command_handlers = {} self.ui_components = {} self._setup_ui() def _setup_ui(self): """设置UI组件""" self.status_bar = StatusBar() self.command_input = CommandInput() self.content_area = MainContentArea() # 注册命令处理器 self.register_command_handler("chat", self._handle_chat_command) self.register_command_handler("debate", self._handle_debate_command) self.register_command_handler("wiki", self._handle_wiki_command) self.register_command_handler("help", self._handle_help_command) def register_command_handler(self, command: str, handler: Callable): """注册命令处理器""" self.command_handlers[command] = handler async def handle_command(self, command: str, args: List[str] = None): """处理命令""" args = args or [] parts = command.split() main_command = parts[0].lower() if main_command in self.command_handlers: try: self.status_bar.start_operation(f"Executing {main_command}...") # 执行命令 result = await self.command_handlers[main_command](parts[1:]) # 更新状态 self.status_bar.finish_operation(f"{main_command} completed") # 更新内容区域 if result and "view" in result: self.content_area.switch_view(result["view"], result.get("data")) except Exception as e: self.status_bar.finish_operation(f"Error: {str(e)}") else: self.status_bar.finish_operation(f"Unknown command: {main_command}") async def _handle_chat_command(self, args: List[str]) -> Dict[str, Any]: """处理聊天命令""" return {"view": "chat", "data": {"mode": "interactive"}} async def _handle_debate_command(self, args: List[str]) -> Dict[str, Any]: """处理辩论命令""" return {"view": "debate", "data": {"topic": " ".join(args)}} async def _handle_wiki_command(self, args: List[str]) -> Dict[str, Any]: """处理Wiki命令""" return {"view": "wiki", "data": {"query": " ".join(args)}} async def _handle_help_command(self, args: List[str]) -> Dict[str, Any]: """处理帮助命令""" return {"view": "help", "data": {"command": args[0] if args else None}}

3. 异步事件处理系统

from typing import Dict, List, Callable, Any from enum import Enum import asyncio from dataclasses import dataclass class EventType(Enum): """事件类型枚举""" COMMAND_EXECUTED = "command_executed" VIEW_CHANGED = "view_changed" STATUS_UPDATED = "status_updated" ERROR_OCCURRED = "error_occurred" DATA_LOADED = "data_loaded" USER_INPUT = "user_input" @dataclass class UIEvent: """UI事件""" type: EventType data: Any = None source: str = None timestamp: float = None class EventBus: """事件总线""" def __init__(self): self._subscribers: Dict[EventType, List[Callable]] = {} self._event_history: List[UIEvent] = [] self._max_history = 100 def subscribe(self, event_type: EventType, callback: Callable[[UIEvent], None]): """订阅事件""" if event_type not in self._subscribers: self._subscribers[event_type] = [] self._subscribers[event_type].append(callback) def unsubscribe(self, event_type: EventType, callback: Callable[[UIEvent], None]): """取消订阅事件""" if event_type in self._subscribers: try: self._subscribers[event_type].remove(callback) except ValueError: pass # 回调不存在 async def publish(self, event: UIEvent): """发布事件""" if event.timestamp is None: import time event.timestamp = time.time() # 添加到历史记录 self._event_history.append(event) if len(self._event_history) > self._max_history: self._event_history.pop(0) # 通知订阅者 if event.type in self._subscribers: tasks = [] for callback in self._subscribers[event.type]: if asyncio.iscoroutinefunction(callback): tasks.append(callback(event)) else: # 同步回调转换为异步 tasks.append(self._run_sync_callback(callback, event)) if tasks: await asyncio.gather(*tasks, return_exceptions=True) async def _run_sync_callback(self, callback: Callable, event: UIEvent): """运行同步回调""" try: callback(event) except Exception as e: print(f"Error in event callback: {e}") def get_recent_events(self, count: int = 10) -> List[UIEvent]: """获取最近的事件""" return self._event_history[-count:] class AsyncEventHandler: """异步事件处理器""" def __init__(self, event_bus: EventBus): self.event_bus = event_bus self._setup_handlers() def _setup_handlers(self): """设置事件处理器""" self.event_bus.subscribe(EventType.COMMAND_EXECUTED, self._on_command_executed) self.event_bus.subscribe(EventType.VIEW_CHANGED, self._on_view_changed) self.event_bus.subscribe(EventType.ERROR_OCCURRED, self._on_error_occurred) self.event_bus.subscribe(EventType.DATA_LOADED, self._on_data_loaded) async def _on_command_executed(self, event: UIEvent): """处理命令执行事件""" command_data = event.data print(f"Command executed: {command_data}") # 可以在这里添加日志记录、统计等逻辑 # 发布状态更新事件 await self.event_bus.publish(UIEvent( type=EventType.STATUS_UPDATED, data={"message": f"Command {command_data.get('command')} completed"}, source="command_handler" )) async def _on_view_changed(self, event: UIEvent): """处理视图变化事件""" view_data = event.data print(f"View changed to: {view_data.get('view')}") # 可以在这里添加视图切换动画、状态保存等逻辑 async def _on_error_occurred(self, event: UIEvent): """处理错误事件""" error_data = event.data print(f"Error occurred: {error_data.get('message')}") # 可以在这里添加错误日志、用户通知等逻辑 # 发布状态更新事件 await self.event_bus.publish(UIEvent( type=EventType.STATUS_UPDATED, data={"message": f"Error: {error_data.get('message')}", "error": True}, source="error_handler" )) async def _on_data_loaded(self, event: UIEvent): """处理数据加载事件""" data_info = event.data print(f"Data loaded: {data_info}") # 可以在这里添加数据缓存、UI更新等逻辑 class KeyboardManager: """键盘输入管理器""" def __init__(self, event_bus: EventBus): self.event_bus = event_bus self._key_bindings = {} def bind_key(self, key: str, action: str, description: str = ""): """绑定键盘快捷键""" self._key_bindings[key] = { "action": action, "description": description } async def handle_key_press(self, key: str): """处理按键事件""" if key in self._key_bindings: binding = self._key_bindings[key] # 发布用户输入事件 await self.event_bus.publish(UIEvent( type=EventType.USER_INPUT, data={ "type": "key_press", "key": key, "action": binding["action"] }, source="keyboard_manager" )) def get_key_bindings(self) -> Dict[str, Dict[str, str]]: """获取所有键绑定""" return self._key_bindings # 使用示例 async def setup_tui_event_system(): """设置TUI事件系统""" # 创建事件总线 event_bus = EventBus() # 创建事件处理器 event_handler = AsyncEventHandler(event_bus) # 创建键盘管理器 keyboard_manager = KeyboardManager(event_bus) # 绑定快捷键 keyboard_manager.bind_key("ctrl+c", "quit", "Exit application") keyboard_manager.bind_key("ctrl+h", "help", "Show help") keyboard_manager.bind_key("ctrl+l", "clear", "Clear screen") # 模拟事件发布 await event_bus.publish(UIEvent( type=EventType.COMMAND_EXECUTED, data={"command": "debate start AI ethics", "success": True}, source="cli" )) await event_bus.publish(UIEvent( type=EventType.VIEW_CHANGED, data={"view": "debate", "topic": "AI ethics"}, source="ui_manager" )) return event_bus, keyboard_manager

最佳实践与规范

KISS原则应用

  • 简洁的命令接口：CLI命令保持简洁直观，避免复杂的参数组合
  • 清晰的TUI布局：界面布局清晰，功能区域明确分离
  • 最小化配置：合理的默认配置，减少用户配置负担

DRY原则应用

  • 公共UI组件：创建可复用的UI组件库，避免重复开发
  • 统一的命令模式：所有命令遵循统一的参数解析和验证模式
  • 共享的事件处理：事件系统和状态管理在CLI和TUI间共享

YAGNI原则应用

  • 按需功能实现：只实现用户实际需要的功能
  • 渐进式增强：从基础功能开始，逐步添加高级特性
  • 避免过度设计：不需要的复杂功能暂不实现
