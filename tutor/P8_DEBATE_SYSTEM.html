模块概述

模块职责

P8辩论系统是DAIP-LIVE中支持多角色智能体进行结构化辩论和协作讨论的核心模块。它允许不同观点的AI智能体就特定议题进行辩论，通过角色分配、观点对立、逻辑推理和最终总结，为用户提供多角度的思考和分析。

核心功能

  • 多角色辩论：支持正反方角色的智能体参与辩论
  • 模型分配：为不同角色分配不同的AI模型和参数
  • 辩论流程控制：管理辩论的发言顺序和时间控制
  • 观点聚合：收集和整理各方的观点和论据
  • 辩论记录：完整记录辩论过程和结果
  • 质量评估：对辩论质量和逻辑性进行评估

架构设计

核心组件

1. EnhancedDebateManager

增强型辩论管理器，负责协调整个辩论流程，管理参与者、控制发言顺序、记录辩论过程。

2. DebateParticipant

辩论参与者基类，定义了参与者的基本行为接口，支持正方、反方、观察者等不同角色。

3. DebateSession

辩论会话管理，维护辩论的状态、历史记录和参与者信息。

4. RoleBasedModelAssigner

基于角色的模型分配器，为不同的辩论角色分配最适合的AI模型和参数。

5. DebateEvaluator

辩论质量评估器，对辩论的逻辑性、论据质量进行评估和打分。

# 将大接口拆分为多个小接口
class DebateSpeaker(ABC):
    @abstractmethod

        pass

class DebateListener(ABC):
    @abstractmethod

        pass

class DebateEvaluator(ABC):
    @abstractmethod

        pass

class DebateModerator(ABC):
    @abstractmethod

依赖倒置原则实现示例 Python 
class DebateManager:
    """高层模块依赖抽象，不依赖具体实现"""

    def __init__(self,
                 moderator: DebateModerator,
                 recorder: DebateRecorder,
                 evaluator: DebateEvaluator):
        self.moderator = moderator  # 依赖抽象
        self.recorder = recorder    # 依赖抽象
        self.evaluator = evaluator  # 依赖抽象

    async def conduct_debate(self, topic: str, participants: List[DebateSpeaker]) -> DebateResult:
        # 主持人管理流程
        await self.moderator.moderate_session(topic, participants)

        # 记录器记录过程
        for participant in participants:
            statement = await participant.speak(topic, self.get_context())
            await self.recorder.record_statement(participant.get_role(), statement)

        # 评估器评估质量
        scores = await self.evaluator.evaluate_session(self.recorder.get_transcript())

        return DebateResult(transcript=self.recorder.get_transcript(), scores=scores)

3. 辩论质量评估器

from typing import List, Dict, Any import re from collections import Counter class DebateMetrics: """辩论质量指标""" def __init__(self): self.metrics = { "logical_consistency": 0.0, "argument_quality": 0.0, "evidence_support": 0.0, "relevance": 0.0, "clarity": 0.0, "originality": 0.0 } class DebateEvaluator: """辩论质量评估器""" def __init__(self, model_provider: 'ModelProvider'): self.model_provider = model_provider self.evaluation_criteria = self._load_evaluation_criteria() async def evaluate_session(self, session: DebateSession) -> Dict[str, Any]: """评估整个辩论会话""" evaluation_results = {} # 按参与者评估 for participant in session.participants: participant_score = await self._evaluate_participant(session, participant) evaluation_results[participant.role.value] = participant_score # 整体辩论质量评估 overall_score = await self._evaluate_overall_quality(session) evaluation_results["overall"] = overall_score # 计算辩论胜者（基于得分） winner = self._determine_winner(evaluation_results) evaluation_results["winner"] = winner return evaluation_results async def _evaluate_participant(self, session: DebateSession, participant: 'DebateParticipant') -> Dict[str, Any]: """评估单个参与者的表现""" participant_contributions = [ c for c in session.contributions if c.speaker_role == participant.role ] if not participant_contributions: return {"total_score": 0.0, "metrics": DebateMetrics().metrics} # 评估各项指标 metrics = DebateMetrics() # 1. 逻辑一致性 metrics.metrics["logical_consistency"] = await self._evaluate_logical_consistency( participant_contributions ) # 2. 论据质量 metrics.metrics["argument_quality"] = await self._evaluate_argument_quality( participant_contributions ) # 3. 证据支持 metrics.metrics["evidence_support"] = await self._evaluate_evidence_support( participant_contributions ) # 4. 相关性 metrics.metrics["relevance"] = await self._evaluate_relevance( session.topic, participant_contributions ) # 5. 清晰度 metrics.metrics["clarity"] = await self._evaluate_clarity( participant_contributions ) # 6. 原创性 metrics.metrics["originality"] = await self._evaluate_originality( participant_contributions ) # 计算总分 total_score = sum(metrics.metrics.values()) / len(metrics.metrics) return { "total_score": total_score, "metrics": metrics.metrics, "contribution_count": len(participant_contributions), "average_length": self._calculate_average_length(participant_contributions) } async def _evaluate_logical_consistency(self, contributions: List[DebateContribution]) -> float: """评估逻辑一致性""" if len(contributions) < 2: return 0.8 # 默认分数 consistency_scores = [] # 使用AI模型评估贡献之间的逻辑一致性 for i in range(len(contributions) - 1): current = contributions[i] next_contrib = contributions[i + 1] prompt = f""" 请评估以下两个陈述之间的逻辑一致性（0-1分）： 陈述1：{current.content} 陈述2：{next_contrib.content} 评分标准： 1.0 = 完全一致，逻辑连贯 0.8 = 基本一致，有轻微矛盾 0.6 = 部分一致，存在明显矛盾 0.4 = 不一致，逻辑混乱 0.0 = 完全不一致 请只返回一个数字分数： """ try: response = await self.model_provider.generate_response(prompt) score = float(response.strip()) consistency_scores.append(score) except: consistency_scores.append(0.5) # 默认分数 return sum(consistency_scores) / len(consistency_scores) async def _evaluate_argument_quality(self, contributions: List[DebateContribution]) -> float: """评估论据质量""" quality_scores = [] for contribution in contributions: # 分析论据结构的完整性 has_claim = self._has_claim_statement(contribution.content) has_reasoning = self._has_reasoning(contribution.content) has_evidence = self._has_evidence_reference(contribution.content) # 计算结构完整性分数 structure_score = (has_claim + has_reasoning + has_evidence) / 3.0 # 分析论证深度 depth_score = self._analyze_argument_depth(contribution.content) # 综合分数 quality_score = (structure_score * 0.6 + depth_score * 0.4) quality_scores.append(quality_score) return sum(quality_scores) / len(quality_scores) def _has_claim_statement(self, text: str) -> float: """检测是否包含明确的论点陈述""" claim_indicators = ["我认为", "我的观点是", "显然", "毫无疑问", "事实证明"] return any(indicator in text for indicator in claim_indicators) def _has_reasoning(self, text: str) -> float: """检测是否包含推理过程""" reasoning_indicators = ["因为", "所以", "因此", "由于", "这意味着"] return any(indicator in text for indicator in reasoning_indicators) def _has_evidence_reference(self, text: str) -> float: """检测是否引用证据""" evidence_indicators = ["数据表明", "研究显示", "根据", "举例来说", "比如"] return any(indicator in text for indicator in evidence_indicators) def _analyze_argument_depth(self, text: str) -> float: """分析论证深度""" # 基于文本长度和复杂度的简单评估 word_count = len(text.split()) sentence_count = len(re.split(r'[。！？]', text)) if sentence_count == 0: return 0.0 avg_words_per_sentence = word_count / sentence_count # 深度评分 if word_count > 200 and avg_words_per_sentence > 15: return 1.0 elif word_count > 100 and avg_words_per_sentence > 10: return 0.8 elif word_count > 50: return 0.6 else: return 0.4 async def _evaluate_evidence_support(self, contributions: List[DebateContribution]) -> float: """评估证据支持强度""" evidence_scores = [] for contribution in contributions: # 检测证据类型 has_statistical = bool(re.search(r'\d+%', contribution.content)) has_expert_opinion = any(term in contribution.content for term in ["专家", "学者", "研究"]) has_case_study = any(term in contribution.content for term in ["案例", "实例", "例子"]) has_source_reference = bool(re.search(r'[文中引用].*?[文尾引用]', contribution.content)) # 计算证据强度分数 evidence_strength = ( (has_statistical * 0.3) + (has_expert_opinion * 0.3) + (has_case_study * 0.2) + (has_source_reference * 0.2) ) evidence_scores.append(evidence_strength) return sum(evidence_scores) / len(evidence_scores) async def _evaluate_relevance(self, topic: str, contributions: List[DebateContribution]) -> float: """评估内容与主题的相关性""" relevance_scores = [] for contribution in contributions: # 使用AI模型评估相关性 prompt = f""" 请评估以下内容与主题的相关性（0-1分）： 主题：{topic} 内容：{contribution.content} 评分标准： 1.0 = 高度相关，直接讨论主题 0.8 = 相关，围绕主题展开 0.6 = 部分相关，有所偏离 0.4 = 不太相关，偏离主题 0.0 = 完全无关 请只返回一个数字分数： """ try: response = await self.model_provider.generate_response(prompt) score = float(response.strip()) relevance_scores.append(score) except: relevance_scores.append(0.5) # 默认分数 return sum(relevance_scores) / len(relevance_scores) def _determine_winner(self, evaluation_results: Dict[str, Any]) -> Optional[str]: """确定辩论胜者""" participant_scores = { role: data["total_score"] for role, data in evaluation_results.items() if role not in ["overall", "winner"] } if not participant_scores: return None # 如果分差很小，认为是平局 scores = list(participant_scores.values()) max_score = max(scores) min_score = min(scores) if max_score - min_score < 0.1: # 分差小于0.1视为平局 return "draw" # 找出最高分者 winners = [role for role, score in participant_scores.items() if score == max_score] if len(winners) == 1: return winners[0] else: return "draw" # 多个同分也是平局

最佳实践与规范

KISS原则应用

  • 简单的辩论流程：核心辩论流程保持线性化，易于理解和跟踪
  • 清晰的角色定义：每个辩论角色的职责和边界清晰明确
  • 最小化的状态管理：只在必要的地方维护状态信息

DRY原则应用

  • 公共评估逻辑：辩论质量评估的公共逻辑提取为独立组件
  • 可复用的参与者基类：所有参与者继承自统一的基础类
  • 统一的会话管理：辩论会话的管理逻辑统一处理

YAGNI原则应用

  • 按需角色实现：只实现当前需要的辩论角色
  • 渐进式功能扩展：从基础辩论功能开始，逐步添加高级特性
  • 避免过度复杂化：不需要的功能暂不实现
