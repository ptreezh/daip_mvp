需求一致性检测目标

需求一致性检测旨在确保需求文档的质量，避免因需求不一致导致的开发问题。

需求内部一致性

  • 需求描述前后一致
  • 验收标准与需求描述匹配
  • 功能边界清晰

需求间一致性

  • 需求之间无冲突
  • 依赖关系明确
  • 优先级合理

需求与设计一致性

  • 设计能够满足需求
  • 实现与需求描述一致
  • 测试覆盖所有需求

检测机制设计

建立系统化的检测机制，确保需求一致性的全面覆盖。

// 需求ID唯一性检查
function checkUniqueRequirementIds(requirements) {
  const ids = requirements.map(req => req.id);
  const duplicates = ids.filter((id, index) => 
    ids.indexOf(id) !== index
  );
  
  if (duplicates.length > 0) {
    return {
      valid: false,
      errors: [`发现重复的需求ID: ${duplicates.join(', ')}`]
    };
  }
  
  return { valid: true };
}

// 验收标准完整性检查
function checkAcceptanceCriteriaCompleteness(requirement) {
  const errors = [];
  
  if (!requirement.acceptanceCriteria || 
      requirement.acceptanceCriteria.length === 0) {
    errors.push('缺少验收标准');
  } else {
    requirement.acceptanceCriteria.forEach((criteria, index) => {
      if (!criteria || criteria.trim().length === 0) {
        errors.push(`验收标准第${index + 1}项为空`);
      } else if (criteria.length < 10) {
        errors.push(`验收标准第${index + 1}项过于简单`);
      }
    });
  }
  
  return {
    valid: errors.length === 0,
    errors
  };
}

// 依赖关系有效性检查
function checkDependencyValidity(requirement, allRequirements) {
  const errors = [];
  const existingIds = allRequirements.map(req => req.id);
  
  if (requirement.dependencies) {
    requirement.dependencies.forEach(depId => {
      if (!existingIds.includes(depId)) {
        errors.push(`依赖的需求ID "${depId}" 不存在`);
      }
    });
  }
  
  return {
    valid: errors.length === 0,
    errors
  };
}

需求一致性检测工具 
class RequirementConsistencyChecker {
  constructor() {
    this.checkRules = [
      this.checkUniqueId,
      this.checkAcceptanceCriteria,
      this.checkDependencies,
      this.checkContentCompleteness
    ];
  }
  
  // 检查需求ID唯一性
  checkUniqueId(requirements) {
    const ids = requirements.map(req => req.id);
    const duplicates = ids.filter((id, index) => 
      ids.indexOf(id) !== index
    );
    
    if (duplicates.length > 0) {
      return {
        rule: '唯一性检查',
        passed: false,
        message: `发现重复的需求ID: ${duplicates.join(', ')}`
      };
    }
    
    return {
      rule: '唯一性检查',
      passed: true,
      message: '所有需求ID唯一'
    };
  }
  
  // 检查验收标准完整性
  checkAcceptanceCriteria(requirement) {
    if (!requirement.acceptanceCriteria || 
        requirement.acceptanceCriteria.length === 0) {
      return {
        rule: '验收标准检查',
        passed: false,
        message: '缺少验收标准'
      };
    }
    
    for (let i = 0; i < requirement.acceptanceCriteria.length; i++) {
      const criteria = requirement.acceptanceCriteria[i];
      if (!criteria || criteria.trim().length === 0) {
        return {
          rule: '验收标准检查',
          passed: false,
          message: `验收标准第${i + 1}项为空`
        };
      }
      if (criteria.length < 10) {
        return {
          rule: '验收标准检查',
          passed: false,
          message: `验收标准第${i + 1}项过于简单`
        };
      }
    }
    
    return {
      rule: '验收标准检查',
      passed: true,
      message: '验收标准完整'
    };
  }
  
  // 检查依赖关系有效性
  checkDependencies(requirement, allRequirements) {
    if (!requirement.dependencies || 
        requirement.dependencies.length === 0) {
      return {
        rule: '依赖关系检查',
        passed: true,
        message: '无依赖关系'
      };
    }
    
    const existingIds = allRequirements.map(req => req.id);
    for (const depId of requirement.dependencies) {
      if (!existingIds.includes(depId)) {
        return {
          rule: '依赖关系检查',
          passed: false,
          message: `依赖的需求ID "${depId}" 不存在`
        };
      }
    }
    
    return {
      rule: '依赖关系检查',
      passed: true,
      message: '依赖关系有效'
    };
  }
  
  // 检查内容完整性
  checkContentCompleteness(requirement) {
    const issues = [];
    
    if (!requirement.title || requirement.title.trim().length < 5) {
      issues.push('需求标题过短');
    }
    
    if (!requirement.description || requirement.description.trim().length < 20) {
      issues.push('需求描述过于简短');
    }
    
    // 检查是否有模糊词汇
    const fuzzyWords = ['大概', '可能', '差不多', '左右', '等等'];
    for (const word of fuzzyWords) {
      if (requirement.description && requirement.description.includes(word)) {
        issues.push(`需求描述中包含模糊词汇: "${word}"`);
      }
    }
    
    if (issues.length > 0) {
      return {
        rule: '内容完整性检查',
        passed: false,
        message: issues.join('; ')
      };
    }
    
    return {
      rule: '内容完整性检查',
      passed: true,
      message: '内容完整'
    };
  }
  
  // 执行单项检查
  checkRequirement(requirement, allRequirements) {
    const results = [];
    
    for (const rule of this.checkRules) {
      try {
        const result = rule.call(this, requirement, allRequirements);
        results.push(result);
      } catch (error) {
        results.push({
          rule: rule.name,
          passed: false,
          message: `检查出错: ${error.message}`
        });
      }
    }
    
    return results;
  }
  
  // 执行全套检查
  checkAllRequirements(requirements) {
    const report = {
      total: requirements.length,
      passed: 0,
      failed: 0,
      details: []
    };
    
    for (const requirement of requirements) {
      const checkResults = this.checkRequirement(requirement, requirements);
      const requirementPassed = checkResults.every(result => result.passed);
      
      if (requirementPassed) {
        report.passed++;
      } else {
        report.failed++;
      }
      
      report.details.push({
        requirementId: requirement.id,
        requirementTitle: requirement.title,
        passed: requirementPassed,
        checks: checkResults
      });
    }
    
    return report;
  }
  
  // 生成检测报告
  generateReport(report) {
    let reportText = `需求一致性检测报告\n`;
    reportText += `==================\n\n`;
    reportText += `检测时间: ${new Date().toLocaleString()}\n`;
    reportText += `总需求数量: ${report.total}\n`;
    reportText += `通过检测: ${report.passed}\n`;
    reportText += `存在问题: ${report.failed}\n\n`;
    
    if (report.failed > 0) {
      reportText += `❌ 发现问题的需求数量: ${report.failed}\n\n`;
      reportText += `详细问题列表:\n`;
      
      for (const detail of report.details) {
        if (!detail.passed) {
          reportText += `\n需求ID: ${detail.requirementId}\n`;
          reportText += `需求标题: ${detail.requirementTitle}\n`;
          
          for (const check of detail.checks) {
            if (!check.passed) {
              reportText += `  - ${check.rule}: ${check.message}\n`;
            }
          }
        }
      }
    } else {
      reportText += `✅ 所有需求均通过一致性检测\n`;
    }
    
    return reportText;
  }
}

// 使用示例
const checker = new RequirementConsistencyChecker();

// 示例需求数据
const sampleRequirements = [
  {
    id: 'REQ-001',

    acceptanceCriteria: [
      '用户输入有效的用户名和密码后能够成功登录',
      '输入无效凭据时显示相应的错误信息',
      '密码字段应隐藏显示'
    ],
    dependencies: []
  },
  {
    id: 'REQ-002',
    title: '用户注册功能',
    description: '新用户应能够注册账户',
    acceptanceCriteria: [
      '用户填写必要信息后能够成功注册',
      '重复用户名应提示已存在',
      '注册成功后自动登录'
    ],
    dependencies: ['REQ-001']
  },
  {
    id: 'REQ-001', // 重复ID
    title: '商品搜索功能',
    description: '用户应能够搜索商品',
    acceptanceCriteria: [], // 缺少验收标准
    dependencies: ['REQ-999'] // 不存在的依赖
  }
];

// 执行检测
const report = checker.checkAllRequirements(sampleRequirements);
const reportText = checker.generateReport(report);
console.log(reportText);
