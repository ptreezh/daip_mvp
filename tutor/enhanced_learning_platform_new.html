# AI规范化编程示例
class UserRepository:
    """用户数据访问层 - 遵循单一职责原则"""
    
    def __init__(self, database_connection):
        self._db = database_connection
    
    def get_user_by_id(self, user_id: int) -> Optional[User]:
        """根据ID获取用户信息"""
        try:
            query = "SELECT * FROM users WHERE id = ?"
            result = self._db.execute(query, (user_id,))
            return self._map_row_to_user(result.fetchone())
        except Exception as e:
            logger.error(f"获取用户失败: {e}")
            return None
    
    def _map_row_to_user(self, row) -> User:
        """将数据库行映射为用户对象"""
        return User(
            id=row[0],
            name=row[1],
            email=row[2],
            created_at=row[3]
        )

设计原理讲解

单一职责原则 (SRP)

UserRepository类只负责用户数据的访问，不包含业务逻辑或界面代码

异常处理

统一的异常处理机制，记录错误信息并返回安全的默认值

数据映射

私有方法负责将数据库行转换为领域对象，保持代码整洁

✅ 代码质量检查
• 符合SOLID原则
• 异常处理完整
• 代码可读性良好
• 遵循命名规范
