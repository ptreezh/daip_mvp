# 多模型辩论系统核心实现
import asyncio
from typing import List, Dict, Any
from dataclasses import dataclass
from enum import Enum

class DebateRole(Enum):
    """辩论角色定义"""
    PROPONENT = "proponent"      # 支持方
    OPPONENT = "opponent"        # 反对方
    MODERATOR = "moderator"      # 主持人
    EVALUATOR = "evaluator"      # 评估者

@dataclass
class DebateMessage:
    """辩论消息"""
    model_id: str
    role: DebateRole
    content: str
    timestamp: float
    round: int

class MultiModelDebate:
    """多模型辩论系统核心类"""
    
    def __init__(self, models: List[str]):
        self.models = models
        self.debate_history: List[DebateMessage] = []
        self.current_round = 0
        self.max_rounds = 5
        
    async def start_debate(self, topic: str) -> Dict[str, Any]:
        """启动辩论"""
        print(f"🎯 开始辩论主题: {topic}")
        
        # 初始化辩论
        await self._initialize_debate(topic)
        
        # 执行多轮辩论
        for round_num in range(1, self.max_rounds + 1):
            await self._execute_round(topic, round_num)
            
        # 评估结果
        final_result = await self._evaluate_debate(topic)
        
        return {
            "topic": topic,
            "rounds": self.current_round,
            "final_result": final_result,
            "debate_history": self.debate_history
        }
    
    async def _initialize_debate(self, topic: str):
        """初始化辩论"""
        self.debate_history.clear()
        self.current_round = 0
        
        # 为每个模型分配角色
        model_roles = {
            model: DebateRole.MODERATOR if i == 0 else
                   DebateRole.PROPONENT if i % 2 == 1 else DebateRole.OPPONENT
            for i, model in enumerate(self.models)
        }
        
        # 发布辩论主题
        opening_message = f"请针对以下主题进行深入辩论: {topic}"
        await self._broadcast_message("system", DebateRole.MODERATOR, opening_message)
    
    async def _execute_round(self, topic: str, round_num: int):
        """执行单轮辩论"""
        print(f"🔄 执行第 {round_num} 轮辩论")
        
        # 每轮按顺序发言
        for model in self.models:
            # 根据角色生成回应
            response = await self._generate_model_response(model, topic, round_num)
            
            # 添加到辩论历史
            await self._add_to_history(model, response, round_num)
            
            # 短暂延迟，模拟真实对话
            await asyncio.sleep(0.5)
        
        self.current_round = round_num
    
    async def _generate_model_response(self, model_id: str, topic: str, round_num: int) -> str:
        """生成模型回应"""
        # 获取相关历史消息
        relevant_history = [
            msg for msg in self.debate_history[-10:]  # 最近10条消息
        ]
        
        # 构建提示词
        prompt = f"""
        主题: {topic}
        轮次: {round_num}
        你的角色: {self._get_model_role(model_id).value}
        
        辩论历史:
        {chr(10).join([f"{msg.role.value}: {msg.content}" for msg in relevant_history])}
        
        请基于你的角色立场，用专业的角度进行回应：
        """
        
        # 模拟API调用
        # response = await call_model_api(model_id, prompt)
        response = f"[{model_id}] 第{round_num}轮回应: 针对'{topic}'的深入分析..."
        
        return response
    
    def _get_model_role(self, model_id: str) -> DebateRole:
        """获取模型角色"""
        model_index = self.models.index(model_id)
        if model_index == 0:
            return DebateRole.MODERATOR
        elif model_index % 2 == 1:
            return DebateRole.PROPONENT
        else:
            return DebateRole.OPPONENT
    
    async def _add_to_history(self, model_id: str, content: str, round_num: int):
        """添加到辩论历史"""
        message = DebateMessage(
            model_id=model_id,
            role=self._get_model_role(model_id),
            content=content,
            timestamp=asyncio.get_event_loop().time(),
            round=round_num
        )
        self.debate_history.append(message)
    
    async def _evaluate_debate(self, topic: str) -> Dict[str, Any]:
        """评估辩论结果"""
        print("📊 评估辩论结果...")
        
        # 分析辩论内容
        analysis = {
            "total_rounds": self.current_round,
            "participant_count": len(self.models),
            "debate_quality": self._analyze_debate_quality(),
            "consensus_level": self._calculate_consensus(),
            "key_points": self._extract_key_points(),
            "confidence_score": self._calculate_confidence()
        }
        
        return analysis
    
    def _analyze_debate_quality(self) -> float:
        """分析辩论质量"""
        if not self.debate_history:
            return 0.0
        
        # 基于消息数量和轮次计算质量
        total_messages = len(self.debate_history)
        unique_models = len(set(msg.model_id for msg in self.debate_history))
        
        quality_score = min(1.0, (total_messages * unique_models) / (self.max_rounds * len(self.models) * 2))
        return round(quality_score, 2)
    
    def _calculate_consensus(self) -> float:
        """计算共识度"""
        # 简化的共识度计算
        # 实际应用中会使用更复杂的文本相似度算法
        if len(self.debate_history) < 3:
            return 0.0
        
        # 模拟计算
        return round(0.7 + (len(self.debate_history) / 100), 2)
    
    def _extract_key_points(self) -> List[str]:
        """提取关键点"""
        # 从辩论历史中提取关键观点
        key_points = []
        for message in self.debate_history[-5:]:  # 最近5条消息
            if len(message.content) > 50:  # 只考虑较长的消息
                key_points.append(f"[{message.role.value}] {message.content[:100]}...")
        
        return key_points[:3]  # 返回前3个关键点
    
    def _calculate_confidence(self) -> float:
        """计算置信度"""
        # 基于辩论轮次和参与度计算置信度
        if self.current_round == 0:
            return 0.0
        
        participation_rate = len(set(msg.model_id for msg in self.debate_history)) / len(self.models)
        round_completion = self.current_round / self.max_rounds
        
        confidence = (participation_rate * 0.6 + round_completion * 0.4)
        return round(min(1.0, confidence), 2)

# 使用示例
async def main():
    """主函数示例"""
    # 初始化多模型辩论系统
    models = ["gpt-4", "claude-3", "gemini-pro", "llama-2"]
    debate_system = MultiModelDebate(models)
    
    # 启动辩论
    topic = "AI在软件开发中的最佳实践是什么？"
    result = await debate_system.start_debate(topic)
    
    print("🎉 辩论完成!")
    print(f"📊 辩论结果: {result}")
    
    # 分析结果
    print("\n📈 详细分析:")
    print(f"- 辩论轮次: {result['final_result']['total_rounds']}")
    print(f"- 辩论质量: {result['final_result']['debate_quality']}")
    print(f"- 共识度: {result['final_result']['consensus_level']}")
    print(f"- 置信度: {result['final_result']['confidence_score']}")
    print(f"- 关键观点数量: {len(result['final_result']['key_points'])}")

if __name__ == "__main__":
    asyncio.run(main())
