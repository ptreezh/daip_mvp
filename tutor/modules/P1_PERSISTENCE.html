P1 数据持久化模块

SQLite + SQLAlchemy + 会话管理的规范化实现解析

模块概述

P1数据持久化模块是DAIP-LIVE系统的基础设施层，负责所有数据的存储、检索和管理。 该模块完美体现了SOLID原则中的单一职责原则，专注于数据访问的抽象和实现。

P1

核心职责

单一职责原则 DRY原则 清晰原则
  • 数据存储抽象：提供统一的数据访问接口，隐藏底层存储细节
  • 会话管理：管理用户会话数据和系统状态
  • 事务管理：确保数据操作的原子性和一致性
  • 迁移支持：提供数据库结构版本管理
  • 连接池管理：优化数据库连接的性能和资源利用

架构设计

分层架构设计

P1模块采用经典的分层架构，确保各层职责清晰，依赖关系明确：

1

数据访问层 (DAL)

提供CRUD操作的基础接口，使用SQLAlchemy ORM实现数据对象的映射和操作。

2

服务层 (Service Layer)

封装业务逻辑，提供高级的数据操作接口，处理复杂的业务规则。

3

会话管理层

专门负责用户会话的存储和检索，支持会话的创建、更新、删除和查询。

核心设计模式

1. 仓储模式 (Repository Pattern)

抽象数据访问逻辑，使业务层不直接依赖具体的数据库实现。

class BaseRepository: def __init__(self, session: Session): self.session = session def get(self, id: int) -> Optional[ModelType]: return self.session.query(self.model_type).get(id) def get_all(self) -> List[ModelType]: return self.session.query(self.model_type).all() def create(self, obj_data: dict) -> ModelType: obj = self.model_type(**obj_data) self.session.add(obj) self.session.commit() return obj

2. 单元工作模式 (Unit of Work)

管理事务边界，确保一组操作要么全部成功，要么全部失败。

class DatabaseManager: def __enter__(self): self.session = create_session() return self.session def __exit__(self, exc_type, exc_val, exc_tb): if exc_type: self.session.rollback() else: self.session.commit() self.session.close() # 使用示例 with DatabaseManager() as session: user_repo = UserRepository(session) user = user_repo.create({"name": "张三", "email": "zhangsan@example.com"}) # 如果发生异常，自动回滚

3. 工厂模式 (Factory Pattern)

创建数据库连接和服务实例，隐藏复杂的初始化逻辑。

class ServiceFactory: @staticmethod def create_user_service() -> UserService: session = create_session() return UserService(session) @staticmethod def create_conversation_service() -> ConversationService: session = create_session() return ConversationService(session)

3. 会话管理实现

实现完整的用户会话生命周期管理，支持会话的创建、更新、查询和清理。

会话管理实现 Python 
# session_manager.py
from typing import Optional, Dict, Any
from datetime import datetime, timedelta
import uuid
from .database import create_session
from .tables import UserSession

class SessionManager:
    def __init__(self):
        self.session_timeout = timedelta(hours=24)  # 会话超时时间

    def create_session(self, user_id: int, session_data: Dict[str, Any]) -> str:
        """创建新的用户会话"""
        session_id = str(uuid.uuid4())

        with create_session() as db_session:
            user_session = UserSession(
                session_id=session_id,
                user_id=user_id,
                session_data=session_data,
                created_at=datetime.utcnow(),
                last_accessed=datetime.utcnow()
            )
            db_session.add(user_session)
            db_session.commit()

        return session_id

    def get_session(self, session_id: str) -> Optional[Dict[str, Any]]:
        """获取会话数据"""
        with create_session() as db_session:
            user_session = db_session.query(UserSession).filter(
                UserSession.session_id == session_id,
                UserSession.last_accessed > datetime.utcnow() - self.session_timeout
).first() if user_session: # 更新最后访问时间 user_session.last_accessed = datetime.utcnow() db_session.commit() return user_session.session_data return None def update_session(self, session_id: str, session_data: Dict[str, Any]) -> bool: """更新会话数据""" with create_session() as db_session: user_session = db_session.query(UserSession).filter( UserSession.session_id == session_id ).first() if user_session: user_session.session_data.update(session_data) user_session.last_accessed = datetime.utcnow() db_session.commit() return True return False def delete_session(self, session_id: str) -> bool: """删除会话""" with create_session() as db_session: count = db_session.query(UserSession).filter( UserSession.session_id == session_id ).delete() db_session.commit() return count > 0

4. 数据库迁移支持

提供数据库结构版本管理，支持平滑的数据库升级和回滚。

# migration.py from sqlalchemy import text from .database import engine class MigrationManager: def __init__(self): self.migrations = [ {"version": "1.0.0", "description": "初始化数据库结构"}, {"version": "1.1.0", "description": "添加用户表索引"}, {"version": "1.2.0", "description": "添加会话管理表"}, # 更多迁移版本... ] def run_migration(self, target_version: str = None): """运行数据库迁移""" with engine.connect() as conn: current_version = self._get_current_version(conn) for migration in self.migrations: if self._should_apply_migration(migration, current_version, target_version): print(f"应用迁移: {migration['version']} - {migration['description']}") self._apply_migration(conn, migration['version']) conn.commit() def _get_current_version(self, conn) -> str: """获取当前数据库版本""" try: result = conn.execute(text("SELECT version FROM schema_migrations ORDER BY version DESC LIMIT 1")) row = result.fetchone() return row[0] if row else "0.0.0" except: return "0.0.0" def _apply_migration(self, conn, version: str): """应用特定版本的迁移""" if version == "1.0.0": # 创建初始表结构 from .tables import Base Base.metadata.create_all(bind=conn) elif version == "1.1.0": # 添加索引 conn.execute(text("CREATE INDEX IF NOT EXISTS idx_users_email ON users(email)")) elif version == "1.2.0": # 创建会话表 conn.execute(text(""" CREATE TABLE IF NOT EXISTS user_sessions ( id INTEGER PRIMARY KEY AUTOINCREMENT, session_id VARCHAR(255) NOT NULL UNIQUE, user_id INTEGER NOT NULL, session_data TEXT, created_at DATETIME NOT NULL, last_accessed DATETIME NOT NULL, FOREIGN KEY (user_id) REFERENCES users (id) ) """)) # 记录迁移版本 conn.execute(text( "INSERT OR REPLACE INTO schema_migrations (version) VALUES (:version)" ), {"version": version})

5. 事务管理和错误处理

实现完善的事务管理和异常处理机制，确保数据一致性。

# services/base_service.py from typing import Optional, TypeVar, Generic from sqlalchemy.orm import Session from sqlalchemy.exc import SQLAlchemyError, IntegrityError from ..database import create_session import logging logger = logging.getLogger(__name__) ModelType = TypeVar('ModelType') class BaseService(Generic[ModelType]): def __init__(self, session: Session, model_class): self.session = session self.model_class = model_class def create(self, obj_data: dict) -> Optional[ModelType]: """创建对象，包含事务管理和错误处理""" try: obj = self.model_class(**obj_data) self.session.add(obj) self.session.flush() # 预提交，获取ID但不提交事务 logger.info(f"创建 {self.model_class.__name__} 成功: ID={obj.id}") return obj except IntegrityError as e: self.session.rollback() logger.error(f"创建 {self.model_class.__name__} 失败 - 数据完整性错误: {e}") raise ValueError("数据完整性约束违反") except SQLAlchemyError as e: self.session.rollback() logger.error(f"创建 {self.model_class.__name__} 失败 - 数据库错误: {e}") raise RuntimeError("数据库操作失败") def get_by_id(self, id: int) -> Optional[ModelType]: """根据ID获取对象""" try: return self.session.query(self.model_class).filter( self.model_class.id == id ).first() except SQLAlchemyError as e: logger.error(f"查询 {self.model_class.__name__} 失败: {e}") return None def update(self, id: int, update_data: dict) -> Optional[ModelType]: """更新对象""" try: obj = self.get_by_id(id) if not obj: return None for key, value in update_data.items(): if hasattr(obj, key): setattr(obj, key, value) self.session.flush() logger.info(f"更新 {self.model_class.__name__} 成功: ID={id}") return obj except SQLAlchemyError as e: self.session.rollback() logger.error(f"更新 {self.model_class.__name__} 失败: {e}") raise RuntimeError("数据库操作失败") def delete(self, id: int) -> bool: """删除对象""" try: obj = self.get_by_id(id) if not obj: return False self.session.delete(obj) self.session.flush() logger.info(f"删除 {self.model_class.__name__} 成功: ID={id}") return True except SQLAlchemyError as e: self.session.rollback() logger.error(f"删除 {self.model_class.__name__} 失败: {e}") raise RuntimeError("数据库操作失败")

编程原则体现

S - 单一职责原则 (SRP)

在P1模块中的体现：

  • DatabaseManager：只负责数据库连接和会话管理
  • SessionManager：只负责用户会话的生命周期管理
  • MigrationManager：只负责数据库结构版本管理
  • BaseService：只负责基础的CRUD操作抽象
# 违反单一职责的例子（错误示范） class BadPersistenceModule: def __init__(self): self.connection = None def manage_database(self): """管理数据库连接""" pass def handle_user_logic(self): """处理用户业务逻辑 - 违反SRP""" pass def send_email_notifications(self): """发送邮件通知 - 违反SRP""" pass # 遵循单一职责的正确实现 class DatabaseManager: # 只负责数据库管理 def manage_database(self): pass class UserService: # 只负责用户业务逻辑 def handle_user_logic(self): pass class NotificationService: # 只负责通知功能 def send_email_notifications(self): pass

O - 开闭原则 (OCP)

在P1模块中的体现：

from abc import ABC, abstractmethod class DataStorageInterface(ABC): """数据存储接口 - 开闭原则体现""" @abstractmethod def save(self, data: dict) -> bool: pass @abstractmethod def load(self, key: str) -> Optional[dict]: pass @abstractmethod def delete(self, key: str) -> bool: pass class SQLiteStorage(DataStorageInterface): """SQLite存储实现""" def save(self, data: dict) -> bool: # SQLite实现 pass def load(self, key: str) -> Optional[dict]: # SQLite实现 pass def delete(self, key: str) -> bool: # SQLite实现 pass class PostgreSQLStorage(DataStorageInterface): """PostgreSQL存储实现 - 新扩展，不修改现有代码""" def save(self, data: dict) -> bool: # PostgreSQL实现 pass def load(self, key: str) -> Optional[dict]: # PostgreSQL实现 pass def delete(self, key: str) -> bool: # PostgreSQL实现 pass class StorageFactory: @staticmethod def create_storage(storage_type: str) -> DataStorageInterface: if storage_type == "sqlite": return SQLiteStorage() elif storage_type == "postgresql": return PostgreSQLStorage() else: raise ValueError(f"Unsupported storage type: {storage_type}")

K - 保持简单原则 (KISS)

D - 不要重复原则 (DRY)

在P1模块中的体现：

  • 基础服务类：提取公共的CRUD操作逻辑
  • 通用工具函数：避免重复的数据库操作代码
  • 配置管理：统一的数据库配置和连接管理
  • 错误处理：统一的异常处理和日志记录
# 违反DRY的重复代码（错误示范） class UserService: def create_user(self, user_data): session = create_session() try: user = User(**user_data) session.add(user) session.commit() session.close() except Exception as e: session.rollback() session.close() raise e class ConversationService: def create_conversation(self, conv_data): session = create_session() # 重复代码 try: conv = Conversation(**conv_data) session.add(conv) session.commit() # 重复代码 session.close() # 重复代码 except Exception as e: session.rollback() # 重复代码 session.close() # 重复代码 raise e # 遵循DRY的正确实现 class BaseRepository: # 消除重复代码 def __init__(self, model_class): self.model_class = model_class def create(self, data: dict): return self._execute_with_transaction( lambda session: self._create_in_session(session, data) ) def _execute_with_transaction(self, operation): """统一的事务处理，消除重复代码""" session = create_session() try: result = operation(session) session.commit() return result except Exception as e: session.rollback() raise e finally: session.close() def _create_in_session(self, session, data): """在指定会话中创建对象""" obj = self.model_class(**data) session.add(obj) session.flush() # 预提交获取ID return obj class UserService(BaseRepository): def __init__(self, session=None): if session: self.session = session self._manual_session = True else: super().__init__(User) self._manual_session = False def create_user(self, user_data): if self._manual_session: return self._create_in_session(self.session, user_data) return super().create(user_data) class ConversationService(BaseRepository): def __init__(self): super().__init__(Conversation) def create_conversation(self, conv_data): return super().create(conv_data)
