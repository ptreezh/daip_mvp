1. 向量搜索引擎

基于FAISS构建的高效向量检索系统，支持大规模文档的实时相似度搜索。

class VectorSearchEngine: def __init__(self, dimension: int = 768): self.dimension = dimension self.index = faiss.IndexFlatL2(dimension) self.document_store = {} def add_documents(self, documents: List[Dict]): """添加文档到向量索引""" vectors = [] for doc in documents: vector = self._encode_document(doc['content']) vectors.append(vector) self.document_store[len(self.document_store)] = doc # 批量添加到FAISS索引 vectors_array = np.array(vectors).astype('float32') self.index.add(vectors_array) def search(self, query: str, k: int = 10) -> List[Dict]: """语义搜索""" query_vector = self._encode_document(query).reshape(1, -1) distances, indices = self.index.search(query_vector, k) results = [] for i, (dist, idx) in enumerate(zip(distances[0], indices[0])): if idx != -1: # 有效索引 doc = self.document_store[idx] doc['similarity'] = 1 - dist # 转换为相似度 results.append(doc) return results

关键实现细节

1. FAISS向量索引实现

使用FAISS库构建高效的向量搜索引擎，支持大规模文档的快速检索。

# knowledge/vector_index.py import faiss import numpy as np from typing import List, Dict, Optional from sentence_transformers import SentenceTransformer import pickle import os class FAISSVectorIndex: def __init__(self, model_name: str = 'all-MiniLM-L6-v2', index_path: str = None): """ 初始化向量索引 Args: model_name: 用于文本向量化的模型名称 index_path: 索引文件保存路径 """ self.model = SentenceTransformer(model_name) self.index_path = index_path or "knowledge/vector_index.faiss" self.metadata_path = index_path.replace('.faiss', '.metadata') if index_path else "knowledge/vector_index.metadata" self.dimension = self.model.get_sentence_embedding_dimension() self.index = None self.documents = [] self._load_or_create_index() def _load_or_create_index(self): """加载现有索引或创建新索引""" if os.path.exists(self.index_path): self._load_index() else: self._create_index() def _create_index(self): """创建新的FAISS索引""" self.index = faiss.IndexFlatIP(self.dimension) self.documents = [] def _load_index(self): """加载现有索引""" self.index = faiss.read_index(self.index_path) with open(self.metadata_path, 'rb') as f: self.documents = pickle.load(f) def _save_index(self): """保存索引到磁盘""" # 确保目录存在 os.makedirs(os.path.dirname(self.index_path), exist_ok=True) # 保存FAISS索引 faiss.write_index(self.index, self.index_path) # 保存元数据 with open(self.metadata_path, 'wb') as f: pickle.dump(self.documents, f) def add_documents(self, documents: List[Dict]): """批量添加文档到索引""" new_documents = [] vectors = [] for doc in documents: # 生成文档向量 if 'content' in doc: content = doc['content'] vector = self.model.encode([content])[0] vectors.append(vector) # 添加文档元数据 doc_meta = { 'id': len(self.documents) + len(new_documents), 'title': doc.get('title', ''), 'content': content, 'path': doc.get('path', ''), 'tags': doc.get('tags', []), 'created_at': doc.get('created_at', datetime.utcnow().isoformat()) } new_documents.append(doc_meta) # 添加到FAISS索引 if vectors: vectors_array = np.array(vectors).astype('float32') self.index.add(vectors_array) # 保存到文档列表 self.documents.extend(new_documents) # 更新索引文件 self._save_index() def search(self, query: str, k: int = 10, filters: Dict = None) -> List[Dict]: """语义搜索文档""" # 生成查询向量 query_vector = self.model.encode([query])[0].reshape(1, -1) # 执行搜索 distances, indices = self.index.search(query_vector, min(k, len(self.documents))) results = [] for dist, idx in zip(distances[0], indices[0]): if idx >= 0 and idx < len(self.documents): doc = self.documents[idx].copy() doc['similarity'] = float(1 - dist) # L2距离转换为相似度 doc['index'] = int(idx) # 应用过滤器 if self._apply_filters(doc, filters): results.append(doc) return results[:k] def _apply_filters(self, doc: Dict, filters: Optional[Dict]) -> bool: """应用文档过滤器""" if not filters: return True for key, value in filters.items(): if key == 'tags' and isinstance(value, list): # 标签过滤：至少有一个匹配 if not any(tag in doc.get('tags', []) for tag in value): return False elif key == 'created_after' and doc.get('created_at'): if doc['created_at'] < value: return False elif key == 'created_before' and doc.get('created_at'): if doc['created_at'] > value: return False return True def delete_document(self, doc_id: int) -> bool: """从索引中删除文档""" # FAISS不支持单个删除，需要重建索引 if 0 <= doc_id < len(self.documents): # 移除文档 del self.documents[doc_id] # 重建索引 self._rebuild_index() return True return False def _rebuild_index(self): """重建索引""" self.index = faiss.IndexFlatIP(self.dimension) if self.documents: vectors = [] for doc in self.documents: vector = self.model.encode([doc['content']])[0] vectors.append(vector) vectors_array = np.array(vectors).astype('float32') self.index.add(vectors_array) self._save_index() def get_document_by_id(self, doc_id: int) -> Optional[Dict]: """根据ID获取文档""" if 0 <= doc_id < len(self.documents): return self.documents[doc_id] return None def get_statistics(self) -> Dict: """获取索引统计信息""" return { 'total_documents': len(self.documents), 'index_dimension': self.dimension, 'index_size_mb': os.path.getsize(self.index_path) / (1024 * 1024) if os.path.exists(self.index_path) else 0, 'last_updated': datetime.fromtimestamp(os.path.getmtime(self.index_path)).isoformat() if os.path.exists(self.index_path) else None }

3. 语义搜索引擎实现

结合传统关键词搜索和现代向量搜索，提供智能的文档检索能力。

语义搜索引擎实现 Python 
# knowledge/semantic_search.py
from typing import List, Dict, Optional
import re
import jieba
from .vector_index import FAISSVectorIndex
from ..wiki.models import WikiPage

class SemanticSearchEngine:
    def __init__(self, vector_index: FAISSVectorIndex):
        self.vector_index = vector_index
        self.keyword_index = {}  # 关键词倒排索引
        self._build_keyword_index()
def _build_keyword_index(self): """构建关键词倒排索引""" # 从数据库获取所有Wiki页面 with create_session() as session: pages = session.query(WikiPage).all() for page in pages: # 分词和提取关键词 words = self._extract_keywords(page.title + " " + page.content) for word in words: if word not in self.keyword_index: self.keyword_index[word] = [] self.keyword_index[word].append({ 'page_id': page.id, 'title': page.title, 'path': page.path, 'snippet': self._get_snippet(page.content, word) }) def _extract_keywords(self, text: str) -> List[str]: """提取文本中的关键词""" # 使用jieba中文分词 words = jieba.lcut(text.lower()) # 过滤掉停用词和短词 stopwords = {'的', '了', '在', '是', '我', '有', '和', '就', '不', '人', '都', '一', '一个', '上', '也', '很', '到', '说', '要', '去', '你', '会', '着', '没有', '看', '好', '自己', '这'} keywords = [] for word in words: if len(word) >= 2 and word not in stopwords: keywords.append(word) return list(set(keywords)) def _get_snippet(self, content: str, keyword: str, context_length: int = 100) -> str: """获取包含关键词的文本片段""" try: index = content.lower().find(keyword.lower()) if index == -1: return "" start = max(0, index - context_length) end = min(len(content), index + len(keyword) + context_length) snippet = content[start:end].strip() if snippet: snippet += "..." if end < len(content) else "" snippet = "..." + snippet if start > 0 else snippet return snippet except Exception: return "" def search(self, query: str, mode: str = 'hybrid', limit: int = 10) -> List[Dict]: """ 智能搜索 Args: query: 搜索查询 mode: 搜索模式 ('keyword', 'semantic', 'hybrid') limit: 返回结果数量限制 """ if mode == 'keyword': return self._keyword_search(query, limit) elif mode == 'semantic': return self._semantic_search(query, limit) elif mode == 'hybrid': return self._hybrid_search(query, limit) else: raise ValueError(f"不支持的搜索模式: {mode}") def _keyword_search(self, query: str, limit: int) -> List[Dict]: """关键词搜索""" keywords = self._extract_keywords(query) if not keywords: return [] results = {} for keyword in keywords: if keyword in self.keyword_index: for item in self.keyword_index[keyword]: page_id = item['page_id'] if page_id not in results: results[page_id] = item.copy() results[page_id]['score'] = 0 results[page_id]['matched_keywords'] = [] results[page_id]['score'] += 1 results[page_id]['matched_keywords'].append(keyword) # 按分数排序 sorted_results = sorted(results.values(), key=lambda x: x['score'], reverse=True) return sorted_results[:limit] def _semantic_search(self, query: str, limit: int) -> List[Dict]: """语义搜索""" vector_results = self.vector_index.search(query, k=limit) # 添加搜索类型标识 for result in vector_results: result['search_type'] = 'semantic' result['score'] = result['similarity'] return vector_results def _hybrid_search(self, query: str, limit: int) -> List[Dict]: """混合搜索：结合关键词和语义搜索""" # 分别获取关键词和语义搜索结果 keyword_results = self._keyword_search(query, limit) semantic_results = self._semantic_search(query, limit) # 合并结果 all_results = {} # 添加关键词搜索结果 for result in keyword_results: page_id = result['page_id'] all_results[page_id] = result.copy() all_results[page_id]['keyword_score'] = result['score'] all_results[page_id]['semantic_score'] = 0 all_results[page_id]['search_type'] = 'keyword' # 添加语义搜索结果 for result in semantic_results: page_id = result['page_id'] if page_id in all_results: # 合并分数 all_results[page_id]['semantic_score'] = result['score'] all_results[page_id]['search_type'] = 'hybrid' else: all_results[page_id] = result.copy() all_results[page_id]['keyword_score'] = 0 all_results[page_id]['semantic_score'] = result['score'] all_results[page_id]['search_type'] = 'semantic' # 计算综合分数 for result in all_results.values(): # 关键词搜索权重0.6，语义搜索权重0.4 result['final_score'] = ( result['keyword_score'] * 0.6 + result['semantic_score'] * 0.4 ) # 按综合分数排序 sorted_results = sorted(all_results.values(), key=lambda x: x['final_score'], reverse=True) return sorted_results[:limit] def suggest_keywords(self, partial_query: str, limit: int = 10) -> List[str]: """关键词自动补全""" suggestions = [] for keyword in self.keyword_index.keys(): if keyword.startswith(partial_query.lower()): suggestions.append(keyword) if len(suggestions) >= limit: break return sorted(suggestions, key=lambda x: len(x), reverse=True)[:limit] def get_search_suggestions(self, query: str, limit: int = 5) -> List[Dict]: """获取搜索建议""" suggestions = [] # 获取相关页面 search_results = self.search(query, mode='semantic', limit=limit) for result in search_results: # 提取页面中的相关关键词 page_keywords = self._extract_keywords( result.get('title', '') + ' ' + result.get('content', '')[:500] # 只取前500字符 ) # 过滤掉查询中的词 query_words = set(self._extract_keywords(query)) relevant_keywords = [kw for kw in page_keywords if kw not in query_words] if relevant_keywords: suggestions.append({ 'page_title': result.get('title', ''), 'page_path': result.get('path', ''), 'suggested_keywords': relevant_keywords[:3] }) return suggestions[:limit]

4. 实时协作编辑实现

基于WebSocket的实时协作编辑系统，支持多用户同时编辑和冲突解决。

# wiki/collaborative_wiki.py from typing import Dict, List, Set import asyncio import websockets import json from datetime import datetime from dataclasses import dataclass @dataclass class EditOperation: """编辑操作""" operation: str # 'insert', 'delete', 'replace' position: int content: str user_id: str timestamp: datetime class CollaborativeWiki: def __init__(self): self.active_sessions = {} # 活跃的编辑会话 self.document_states = {} # 文档的当前状态 self.operation_history = {} # 操作历史 self.user_cursors = {} # 用户光标位置 async def handle_websocket_connection(self, websocket, path): """处理WebSocket连接""" # 解析路径获取页面ID page_id = self._extract_page_id(path) if not page_id: await websocket.close(code=4000) return user_id = self._authenticate_user(websocket) if not user_id: await websocket.close(code=4001) return # 创建编辑会话 session_id = self._create_edit_session(page_id, user_id) # 发送当前文档状态 current_content = self._get_current_content(page_id) await websocket.send_text(json.dumps({ 'type': 'document_state', 'content': current_content, 'session_id': session_id })) # 处理消息循环 try: while True: message = await websocket.receive_text() data = json.loads(message) await self._handle_message(websocket, data, page_id, user_id, session_id) except websockets.exceptions.ConnectionClosed: pass finally: # 清理会话 self._cleanup_session(session_id) def _create_edit_session(self, page_id: str, user_id: str) -> str: """创建编辑会话""" session_id = str(uuid.uuid4()) self.active_sessions[session_id] = { 'page_id': page_id, 'user_id': user_id, 'websocket': None, 'created_at': datetime.utcnow() } return session_id async def _handle_message(self, websocket, data: Dict, page_id: str, user_id: str, session_id: str): """处理WebSocket消息""" message_type = data.get('type') if message_type == 'edit_operation': await self._handle_edit_operation( websocket, data, page_id, user_id, session_id ) elif message_type == 'cursor_update': await self._handle_cursor_update( websocket, data, page_id, user_id ) elif message_type == 'user_activity': await self._handle_user_activity( websocket, data, page_id, user_id ) async def _handle_edit_operation(self, websocket, data: Dict, page_id: str, user_id: str, session_id: str): """处理编辑操作""" operation = EditOperation( operation=data['operation'], position=data['position'], content=data['content'], user_id=user_id, timestamp=datetime.utcnow() ) # 应用操作到文档 updated_content = self._apply_operation( self.document_states[page_id]['content'], operation ) # 更新文档状态 self.document_states[page_id]['content'] = updated_content self.document_states[page_id]['last_modified'] = operation.timestamp #
